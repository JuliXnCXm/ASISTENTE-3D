```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de la escalera
num_peldaños = 15
altura_peldaño = 0.2  # metros
profundidad_peldaño = 0.25  # metros
ancho_peldaño = 0.3  # metros
longitud_peldaño = 0.15  # metros
altura_total = num_peldaños * altura_peldaño
ancho_descanso = 1.5
longitud_descanso = 3.0
grosor_descanso = 0.1

# Material de hormigón
material_hormigon = bpy.data.materials.new(name="Hormigon")
material_hormigon.use_nodes = True
principled_bsdf = material_hormigon.node_tree.nodes["Principled BSDF"]
principled_bsdf.inputs["Base Color"].default_value = (0.2, 0.2, 0.2, 1)
principled_bsdf.inputs["Roughness"].default_value = 0.8

# Material de metal
material_metal = bpy.data.materials.new(name="Metal")
material_metal.use_nodes = True
principled_bsdf_metal = material_metal.node_tree.nodes["Principled BSDF"]
principled_bsdf_metal.inputs["Base Color"].default_value = (0.8, 0.8, 0.8, 1)
principled_bsdf_metal.inputs["Metallic"].default_value = 1.0
principled_bsdf_metal.inputs["Roughness"].default_value = 0.2


# Crear peldaños
for i in range(num_peldaños):
    x = i * longitud_peldaño
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, 0, i * altura_peldaño))
    peldaño = bpy.context.object
    peldaño.scale = (ancho_peldaño, profundidad_peldaño, altura_peldaño)
    peldaño.name = f"Peldaño_{i}"
    peldaño.data.materials.append(material_hormigon)

# Crear base de la escalera
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
base = bpy.context.object
base.scale = (ancho_peldaño, profundidad_peldaño, 0.05)
base.name = "Base"
base.data.materials.append(material_hormigon)

# Crear el descanso superior
bpy.ops.mesh.primitive_cube_add(size=1, location=(num_peldaños * longitud_peldaño, 0, altura_total))
descanso = bpy.context.object
descanso.scale = (ancho_descanso, grosor_descanso, longitud_descanso)
descanso.name = "Descanso"
descanso.data.materials.append(material_hormigon)

# Crear baranda
bpy.ops.mesh.primitive_cube_add(size=1, location=(num_peldaños * longitud_peldaño + ancho_descanso/2, 0, altura_total + 0.1))
baranda = bpy.context.object
baranda.scale = (0.05, ancho_descanso, 0.1)
baranda.name = "Baranda"
baranda.data.materials.append(material_metal)

# Mover la baranda a la posición correcta
baranda.location.x = num_peldaños * longitud_peldaño + ancho_descanso/2
baranda.location.y = 0
baranda.location.z = altura_total + 0.1

# Ajustar la vista
bpy.context.scene.camera.location = (num_peldaños * longitud_peldaño + 2, -ancho_descanso, altura_total + 1)
bpy.context.scene.camera.rotation_euler = (math.radians(60), 0, math.radians(45))

# Guardar la escena
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
