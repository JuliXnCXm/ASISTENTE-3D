```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del patio trasero en metros
patio_width = 5.0
patio_depth = 4.0
wall_height = 2.5
wall_thickness = 0.2

# --- Piso de concreto ---
bpy.ops.mesh.primitive_plane_add(size=patio_width, enter_editmode=False, align='WORLD', location=(0, 0, 0))
patio_floor = bpy.context.object
patio_floor.name = "PatioFloor"
patio_floor.scale = (1, patio_depth/patio_width, 1) # Ajustar la escala para la profundidad
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True) # Aplicar la escala
bpy.ops.object.shade_smooth()
bpy.ops.object.select_all(action='DESELECT')
patio_floor.select_set(True)
bpy.context.view_layer.objects.active = patio_floor

# --- Muro de ladrillo ---
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, -patio_depth/2, wall_height/2))
wall = bpy.context.object
wall.name = "BrickWall"
wall.scale = (patio_width/1, 0.1, wall_height/1)
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
bpy.ops.object.shade_smooth()

# Subdividir para simular ladrillos
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.subdivide(number_cuts=10)
bpy.ops.object.mode_set(mode='OBJECT')

# Aplicar un material de ladrillo (simulado)
brick_material = bpy.data.materials.new(name="BrickMaterial")
brick_material.use_nodes = True
bsdf = brick_material.node_tree.nodes["Principled BSDF"]
bsdf.inputs["Base Color"].default_value = (0.6, 0.3, 0.1, 1)  # Color ladrillo
bsdf.inputs["Roughness"].default_value = 0.7
wall.data.materials.append(brick_material)

# --- Opcional: Guardar el archivo .blend ---
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
