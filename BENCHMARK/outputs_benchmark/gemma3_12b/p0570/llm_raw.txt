```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del edificio
width = 20.0  # metros
depth = 15.0  # metros
height_per_floor = 3.0  # metros
total_height = height_per_floor * 5  # metros

# Crear el edificio
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, total_height / 2))
building = bpy.context.object
building.name = "OfficeBuilding"
building.scale = (width / 2, depth / 2, total_height / 2)

# Crear la fachada de muro cortina
num_windows_width = 10
num_windows_depth = 5
window_width = width / num_windows_width
window_depth = depth / num_windows_depth
window_height = 1.0

for floor in range(5):
    for i in range(num_windows_width):
        for j in range(num_windows_depth):
            x = i * window_width - width / 2 + window_width / 2
            y = j * window_depth - depth / 2 + window_depth / 2
            z = floor * height_per_floor + height_per_floor / 2

            bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(x, y, z))
            window = bpy.context.object
            window.name = f"Window_Floor_{floor+1}_{i}_{j}"
            window.scale = (window_width / 2, window_depth / 2, window_height / 2)
            window.rotation_euler = (math.pi / 2, 0, 0) # Rotar para que sea vertical

# Material simple para las ventanas (opcional)
window_material = bpy.data.materials.new(name="WindowMaterial")
window_material.use_nodes = True
bsdf = window_material.node_tree.nodes["Principled BSDF"]
bsdf.inputs["Base Color"].default_value = (0.2, 0.5, 0.8, 1)  # Azul claro
bsdf.inputs["Roughness"].default_value = 0.1
for window in bpy.data.objects:
    if "Window" in window.name:
        if window.data.materials:
            window.data.materials[0] = window_material
        else:
            window.data.materials.append(window_material)

# Material simple para el edificio (opcional)
building_material = bpy.data.materials.new(name="BuildingMaterial")
building_material.use_nodes = True
bsdf = building_material.node_tree.nodes["Principled BSDF"]
bsdf.inputs["Base Color"].default_value = (0.8, 0.8, 0.8, 1)  # Gris claro
bsdf.inputs["Roughness"].default_value = 0.7
building.data.materials.append(building_material)

# Guardar el archivo .blend (opcional)
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
