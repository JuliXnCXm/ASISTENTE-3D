```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del espacio en metros
width = 8
depth = 6
height = 3

# --- Crear el espacio ---
# Crear el suelo
bpy.ops.mesh.primitive_plane_add(size=width, enter_editmode=False, align='WORLD', location=(0, 0, 0))
floor = bpy.context.object
floor.name = "Floor"
floor.scale = (1, depth/width, 1)
floor.data.materials.clear()
material_floor = bpy.data.materials.new(name="Porcelain")
material_floor.use_nodes = True
bsdf = material_floor.node_tree.nodes["Principled BSDF"]
bsdf.inputs["Base Color"].default_value = (0.6, 0.6, 0.6, 1)  # Gris claro
bsdf.inputs["Roughness"].default_value = 0.2
floor.data.materials.append(material_floor)

# Crear las paredes
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(width/2, 0, height/2))
wall_back = bpy.context.object
wall_back.name = "WallBack"
wall_back.scale = (1, depth/2, height/2)
wall_back.location = (width/2, depth/2, height/2)

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, height/2))
wall_left = bpy.context.object
wall_left.name = "WallLeft"
wall_left.scale = (width/2, 1, height/2)
wall_left.location = (0, 0, height/2)

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(width, 0, height/2))
wall_right = bpy.context.object
wall_right.name = "WallRight"
wall_right.scale = (width/2, 1, height/2)
wall_right.location = (width, 0, height/2)

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(width/2, depth, height/2))
wall_front = bpy.context.object
wall_front.name = "WallFront"
wall_front.scale = (1, depth/2, height/2)
wall_front.location = (width/2, depth, height/2)

# --- Crear los muebles ---

# Zona de estar: Sofá
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(1, 2, 0.5))
sofa = bpy.context.object
sofa.name = "Sofa"
sofa.scale = (2, 0.8, 0.5)

# Zona de estar: Estantería
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(3, 2, 1.5))
bookshelf = bpy.context.object
bookshelf.name = "Bookshelf"
bookshelf.scale = (1.5, 0.5, 1)

# Zona de comedor: Mesa
bpy.ops.mesh.primitive_circle_add(radius=1, enter_editmode=False, align='WORLD', location=(4, 4, 0.1))
table = bpy.context.object
table.name = "Table"
table.scale = (1, 1, 0.1)

# Zona de comedor: Sillas (crear una y duplicar)
bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=0.8, enter_editmode=False, align='WORLD', location=(3.5, 4.5, 0.4))
chair = bpy.context.object
chair.name = "Chair"
chair.scale = (1, 0.8, 1)

# Duplicar la silla para las otras posiciones
chair_2 = chair.copy()
chair_2.data = chair.data
chair_2.name = "Chair2"
chair_2.location = (4.5, 4.5, 0.4)

chair_3 = chair.copy()
chair_3.data = chair.data
chair_3.name = "Chair3"
chair_3.location = (3.5, 5.5, 0.4)

chair_4 = chair.copy()
chair_4.data = chair.data
chair_4.name = "Chair4"
chair_4.location = (4.5, 5.5, 0.4)

# --- Crear la luz ---
bpy.ops.object.light_add(type='AREA', radius=1, align='WORLD', location=(width/2, depth/2, height + 1))
light = bpy.context.object
light.name = "AreaLight"
light.data.energy = 500

# Guardar la escena si la variable de entorno BLEND_OUT está definida
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
