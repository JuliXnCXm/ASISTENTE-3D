```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# --- Funciones de utilidad ---
def create_material(name, color, specular=0.1, roughness=0.7):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    bsdf = nodes.get("Principled BSDF")
    if not bsdf:
        bsdf = nodes.new("ShaderNodeBsdfPrincipled")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Specular"].default_value = specular
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

# --- Crear materiales ---
fabric_material = create_material("Fabric_Grey", (0.8, 0.8, 0.8, 1.0))
wood_material = create_material("Wood_Walnut", (0.4, 0.2, 0.1, 1.0), specular=0.05, roughness=0.6)


# --- Crear el sofá ---
sofa_width = 2.0
sofa_depth = 1.0
sofa_height = 0.7
sofa_back_height = 0.5

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, sofa_height/2), scale=(sofa_width, sofa_depth, sofa_height))
sofa_body = bpy.context.object
sofa_body.name = "Sofa_Body"
sofa_body.data.materials.append(fabric_material)

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, sofa_height + sofa_back_height/2), scale=(sofa_width, 0.1, sofa_back_height))
sofa_back = bpy.context.object
sofa_back.name = "Sofa_Back"
sofa_back.data.materials.append(fabric_material)

# --- Crear la mesa de centro ---
table_width = 1.2
table_depth = 0.7
table_height = 0.6

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0.8, 0, table_height/2), scale=(table_width, table_depth, table_height))
table = bpy.context.object
table.name = "Coffee_Table"
table.data.materials.append(wood_material)


# --- Ajustar la posición de los objetos ---
# El sofá está centrado en el origen
# La mesa está ligeramente delante del sofá

# --- Guardar el archivo .blend ---
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
