```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones en metros
ancho_cama = 1.6
largo_cama = 2.0
altura_estructura = 0.4
altura_colchon = 0.25
grosor_madera = 0.05

# Materiales
material_roble = bpy.data.materials.new(name="Roble")
material_roble.use_nodes = True
principled_bsdf = material_roble.node_tree.nodes["Principled BSDF"]
principled_bsdf.inputs["Base Color"].default_value = (0.6, 0.4, 0.2, 1)  # Color roble
principled_bsdf.inputs["Roughness"].default_value = 0.5

material_colchon = bpy.data.materials.new(name="Colchon")
material_colchon.use_nodes = True
principled_bsdf_colchon = material_colchon.node_tree.nodes["Principled BSDF"]
principled_bsdf_colchon.inputs["Base Color"].default_value = (1, 1, 1, 1)  # Blanco
principled_bsdf_colchon.inputs["Roughness"].default_value = 0.8

# Estructura de la cama (Roble)
# Patas
bpy.ops.mesh.primitive_cube_add(size=0.1, location=(ancho_cama/2 - 0.05, largo_cama/2 - 0.05, altura_estructura/2))
pata_1 = bpy.context.object
pata_1.name = "Pata_1"
pata_1.data.materials.append(material_roble)

bpy.ops.mesh.primitive_cube_add(size=0.1, location=(-ancho_cama/2 + 0.05, largo_cama/2 - 0.05, altura_estructura/2))
pata_2 = bpy.context.object
pata_2.name = "Pata_2"
pata_2.data.materials.append(material_roble)

bpy.ops.mesh.primitive_cube_add(size=0.1, location=(ancho_cama/2 - 0.05, -largo_cama/2 + 0.05, altura_estructura/2))
pata_3 = bpy.context.object
pata_3.name = "Pata_3"
pata_3.data.materials.append(material_roble)

bpy.ops.mesh.primitive_cube_add(size=0.1, location=(-ancho_cama/2 + 0.05, -largo_cama/2 + 0.05, altura_estructura/2))
pata_4 = bpy.context.object
pata_4.name = "Pata_4"
pata_4.data.materials.append(material_roble)

# Base de la cama
bpy.ops.mesh.primitive_cube_add(size=ancho_cama, location=(0, 0, altura_estructura + grosor_madera/2), scale=(1, largo_cama/ancho_cama, 1))
base_cama = bpy.context.object
base_cama.name = "Base_Cama"
base_cama.data.materials.append(material_roble)

# Colchón
bpy.ops.mesh.primitive_cube_add(size=ancho_cama, location=(0, 0, altura_estructura + altura_colchon/2), scale=(1, largo_cama/ancho_cama, 1))
colchon = bpy.context.object
colchon.name = "Colchon"
colchon.data.materials.append(material_colchon)

# Ajustar la escala del colchón para que se ajuste a la base
colchon.scale[1] = largo_cama / ancho_cama

# Guardar la escena
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
