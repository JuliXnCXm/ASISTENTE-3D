```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones en metros
pergola_width = 4.0
pergola_depth = 3.0
pergola_height = 2.5

# Dimensiones de la columna
columna_width = 0.2
columna_depth = 0.2
columna_height = pergola_height

# Dimensiones del techo
techo_width = pergola_width
techo_depth = pergola_depth
techo_thickness = 0.1

# Crear columnas
def create_columna(x, y):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, columna_height / 2))
    columna = bpy.context.object
    columna.scale = (columna_width / 2, columna_depth / 2, columna_height / 2)
    columna.name = "Columna"
    return columna

columna1 = create_columna(pergola_width / 2, pergola_depth / 2)
columna2 = create_columna(-pergola_width / 2, pergola_depth / 2)
columna3 = create_columna(-pergola_width / 2, -pergola_depth / 2)
columna4 = create_columna(pergola_width / 2, -pergola_depth / 2)

# Crear techo
bpy.ops.mesh.primitive_plane_add(size=1, location=(0, 0, pergola_height))
techo = bpy.context.object
techo.scale = (techo_width / 2, techo_depth / 2, techo_thickness / 2)
techo.name = "Techo"

    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
