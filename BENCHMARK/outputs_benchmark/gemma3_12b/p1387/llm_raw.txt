```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de la fachada
width = 10
height = 6
depth = 0.2  # Profundidad de la fachada

# Dimensiones de la puerta
door_width = 2
door_height = 2.2
door_depth = depth

# Dimensiones de las ventanas
window_width = 1.5
window_height = 1.0
window_depth = depth

# Crear la fachada
bpy.ops.mesh.primitive_plane_add(size=width, enter_editmode=False, align='WORLD', location=(0, 0, height/2), scale=(1, 1, 1))
facade = bpy.context.object
facade.name = "Facade"
facade.scale[2] = height / 2
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, -depth)})
bpy.ops.object.mode_set(mode='OBJECT')

# Crear la puerta
bpy.ops.mesh.primitive_plane_add(size=door_width, enter_editmode=False, align='WORLD', location=(0, -width/2 + door_width/2, door_height/2), scale=(1, 1, 1))
door = bpy.context.object
door.name = "Door"
door.scale[2] = door_height / 2
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, -door_depth)})
bpy.ops.object.mode_set(mode='OBJECT')

# Crear la ventana izquierda
bpy.ops.mesh.primitive_plane_add(size=window_width, enter_editmode=False, align='WORLD', location=(width/2 - window_width/2 - 0.1, -width/2 + window_width/2, window_height/2), scale=(1, 1, 1))
window_left = bpy.context.object
window_left.name = "WindowLeft"
window_left.scale[2] = window_height / 2
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, -window_depth)})
bpy.ops.object.mode_set(mode='OBJECT')

# Crear la ventana derecha
bpy.ops.mesh.primitive_plane_add(size=window_width, enter_editmode=False, align='WORLD', location=(-width/2 + window_width/2 + 0.1, -width/2 + window_width/2, window_height/2), scale=(1, 1, 1))
window_right = bpy.context.object
window_right.name = "WindowRight"
window_right.scale[2] = window_height / 2
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, -window_depth)})
bpy.ops.object.mode_set(mode='OBJECT')

# Opcional: Guardar el archivo .blend
if 'BLEND_OUT' in os.environ:
    filepath = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=filepath)
```
