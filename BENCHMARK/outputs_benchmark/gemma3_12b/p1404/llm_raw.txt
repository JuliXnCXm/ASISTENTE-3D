```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del muro de contención
longitud = 10.0  # metros
altura = 1.2  # metros
grosor = 0.2  # metros

# Dimensiones del ladrillo
ladrillo_ancho = 0.24  # metros
ladrillo_largo = 0.12  # metros
ladrillo_alto = 0.06  # metros

# Crear el muro de contención
def crear_muro(longitud, altura, grosor, ladrillo_ancho, ladrillo_largo, ladrillo_alto):
    bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
    muro = bpy.context.object
    muro.name = "Muro_Contencion"
    muro.scale = (longitud / 2, grosor / 2, altura / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    # Subdividir el muro para simular ladrillos
    num_ladrillos_ancho = int(grosor / ladrillo_ancho)
    num_ladrillos_largo = int(longitud / ladrillo_largo)
    
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.subdivide(number_cuts=num_ladrillos_ancho - 1)
    bpy.ops.object.mode_set(mode='OBJECT')

    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.subdivide(number_cuts=num_ladrillos_largo - 1)
    bpy.ops.object.mode_set(mode='OBJECT')

    return muro

muro = crear_muro(longitud, altura, grosor, ladrillo_ancho, ladrillo_largo, ladrillo_alto)

# Crear la baranda metálica
def crear_baranda(longitud, altura, grosor):
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=grosor, enter_editmode=False, align='WORLD', location=(0, 0, 0))
    baranda = bpy.context.object
    baranda.name = "Baranda_Metalica"
    baranda.scale = (longitud / 2, 1, 1)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, 0)})
    bpy.ops.object.mode_set(mode='OBJECT')
    
    return baranda

baranda = crear_baranda(longitud, altura, grosor)
baranda.location = (0, 0, altura + 0.05)
```
