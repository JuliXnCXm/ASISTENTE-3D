```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del edificio
longitud = 20.0  # metros
ancho = 15.0  # metros
altura_planta = 3.0  # metros
num_pisos = 4

# Altura total del edificio
altura_total = altura_planta * num_pisos

# Crear el bloque principal del edificio
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, altura_total / 2))
edificio = bpy.context.object
edificio.scale = (longitud / 2, ancho / 2, altura_total / 2)
edificio.name = "Edificio"

# Crear material de hormigón
material_hormigon = bpy.data.materials.new(name="Hormigon")
material_hormigon.use_nodes = True
principled_bsdf = material_hormigon.node_tree.nodes["Principled BSDF"]
principled_bsdf.inputs["Base Color"].default_value = (0.4, 0.4, 0.4, 1)
principled_bsdf.inputs["Roughness"].default_value = 0.7
edificio.data.materials.append(material_hormigon)

# Crear columnas estructurales
diametro_columna = 0.5
num_columnas_longitud = 5
num_columnas_ancho = 3

for i in range(num_columnas_longitud + 1):
    for j in range(num_columnas_ancho + 1):
        x = (i - (num_columnas_longitud / 2)) * (longitud / (num_columnas_longitud))
        y = (j - (num_columnas_ancho / 2)) * (ancho / (num_columnas_ancho))
        
        bpy.ops.mesh.primitive_cylinder_add(radius=diametro_columna / 2, depth=altura_total, enter_editmode=False, align='WORLD', location=(x, y, altura_total / 2))
        columna = bpy.context.object
        columna.name = f"Columna_{i}_{j}"
        
        # Aplicar material de hormigón a la columna
        columna.data.materials.append(material_hormigon)

# Crear ventanas (simples)
ancho_ventana = 1.0
altura_ventana = 1.5
separacion_ventana = 2.0

for i in range(int(longitud / (ancho_ventana + separacion_ventana))):
    for j in range(int(ancho / (ancho_ventana + separacion_ventana))):
        x = -longitud / 2 + (i + 1) * (ancho_ventana + separacion_ventana) - ancho_ventana / 2
        y = -ancho / 2 + (j + 1) * (ancho_ventana + separacion_ventana) - ancho_ventana / 2
        z = altura_planta / 2
        
        bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(x, y, z))
        ventana = bpy.context.object
        ventana.scale = (ancho_ventana / 2, 0.1, altura_ventana / 2)
        ventana.name = f"Ventana_{i}_{j}"
        
        # Crear material transparente para la ventana
        material_transparente = bpy.data.materials.new(name="Transparente")
        material_transparente.use_nodes = True
        principled_bsdf_transparente = material_transparente.node_tree.nodes["Principled BSDF"]
        principled_bsdf_transparente.inputs["Base Color"].default_value = (1, 1, 1, 1)
        principled_bsdf_transparente.inputs["Roughness"].default_value = 0.0
        principled_bsdf_transparente.inputs["Transmission"].default_value = 1.0
        ventana.data.materials.append(material_transparente)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT está definida

    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
