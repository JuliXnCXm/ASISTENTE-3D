```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del terreno en metros
width = 15
length = 10
height = 0

# Crear la malla del terreno
bpy.ops.mesh.plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
terrain = bpy.context.object
terrain.scale = (width / 2, length / 2, 1)
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

# Subdividir la malla para suavizarla
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.subdivide(number_cuts=50)
bpy.ops.object.mode_set(mode='OBJECT')

# Crear un material de césped
material = bpy.data.materials.new(name="GrassMaterial")
material.use_nodes = True
nodes = material.node_tree.nodes
links = material.node_tree.links

# Eliminar el nodo Principled BSDF por defecto
principled_bsdf = nodes.get("Principled BSDF")
if principled_bsdf:
    nodes.remove(principled_bsdf)

# Crear un nodo BSDF de Principio
principled_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
principled_bsdf.location = (200, 0)

# Crear un nodo de textura de color
color_texture = nodes.new(type='ShaderNodeTexColor')
color_texture.location = (-200, 0)
color_texture.color = (0.5, 0.6, 0.3, 1)  # Color verde césped

# Crear un nodo de textura de ruido
noise_texture = nodes.new(type='ShaderNodeTexNoise')
noise_texture.location = (-200, -200)
noise_texture.scale = 2.0
noise_texture.detail = 10

# Conectar los nodos
links.new(noise_texture.outputs['Fac'], principled_bsdf.inputs['Roughness'])
links.new(color_texture.outputs['Color'], principled_bsdf.inputs['Base Color'])
links.new(principled_bsdf.outputs['BSDF'], nodes.new(type='ShaderNodeOutputMaterial').inputs['Surface'])

# Asignar el material al terreno
if terrain.data.materials:
    terrain.data.materials[0] = material
else:
    terrain.data.materials.append(material)

# Ajustar la ubicación del terreno
terrain.location = (0, 0, 0)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT está definida
if 'BLEND_OUT' in os.environ:
    blend_out_path = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
