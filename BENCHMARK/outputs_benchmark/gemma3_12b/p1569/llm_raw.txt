```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del edificio
width = 15
depth = 12
height_per_floor = 3
total_height = height_per_floor * 4

# Materiales
material_building = bpy.data.materials.new(name="BuildingMaterial")
material_building.use_nodes = True
principled_bsdf = material_building.node_tree.nodes["Principled BSDF"]
principled_bsdf.inputs["Base Color"].default_value = (0.8, 0.8, 0.8, 1)  # Gris claro
principled_bsdf.inputs["Roughness"].default_value = 0.7

material_window = bpy.data.materials.new(name="WindowMaterial")
material_window.use_nodes = True
principled_bsdf_window = material_window.node_tree.nodes["Principled BSDF"]
principled_bsdf_window.inputs["Base Color"].default_value = (0.2, 0.4, 0.8, 1)  # Azul claro
principled_bsdf_window.inputs["Roughness"].default_value = 0.2
principled_bsdf_window.inputs["Transmission"].default_value = 1.0
principled_bsdf_window.inputs["Metallic"].default_value = 0.0

# Crear el edificio
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, total_height/2), scale=(width, depth, total_height))
building = bpy.context.object
building.name = "Building"
building.data.materials.append(material_building)

# Crear columnas
column_radius = 0.2
column_height = total_height
num_columns_width = 4
num_columns_depth = 3

for i in range(num_columns_width):
    for j in range(num_columns_depth):
        x = -width/2 + (i + 0.5) * (width / num_columns_width)
        y = -depth/2 + (j + 0.5) * (depth / num_columns_depth)
        bpy.ops.mesh.primitive_cylinder_add(radius=column_radius, depth=column_height, enter_editmode=False, align='WORLD', location=(x, y, column_height/2))
        column = bpy.context.object
        column.name = "Column_" + str(i) + "_" + str(j)
        column.data.materials.append(material_building)

# Crear ventanas
window_width = 2
window_height = 1.5
window_depth = 0.1

for floor in range(4):
    for side in ["X", "Y", "Z", "-X", "-Y", "-Z"]:
        if side == "X":
            start = -width/2 + window_width/2
            end = width/2 - window_width/2
            for i in range(int((end - start) / window_width) + 1):
                x = start + i * window_width
                bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(x, -depth/2 + 0.1, floor * height_per_floor + height_per_floor/2))
                window = bpy.context.object
                window.name = "Window_X_" + str(floor) + "_" + str(i)
                window.scale = (1, window_depth, 1)
                window.data.materials.append(material_window)
        
        elif side == "Y":
            start = -depth/2 + window_width/2
            end = depth/2 - window_width/2
            for i in range(int((end - start) / window_width) + 1):
                y = start + i * window_width
                bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(-width/2 + 0.1, y, floor * height_per_floor + height_per_floor/2))
                window = bpy.context.object
                window.name = "Window_Y_" + str(floor) + "_" + str(i)
                window.scale = (window_depth, 1, 1)
                window.data.materials.append(material_window)
        
        elif side == "Z":
            pass # No hay ventanas en la fachada Z
        
        elif side == "-X":
            pass # No hay ventanas en la fachada -X
        
        elif side == "-Y":
            pass # No hay ventanas en la fachada -Y
        
        elif side == "-Z":
            pass # No hay ventanas en la fachada -Z

# Guardar el archivo
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
