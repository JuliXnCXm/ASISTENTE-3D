```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones del edificio
width = 15
depth = 20
height_per_floor = 3
total_height = height_per_floor * 5
building_height = total_height

# Dimensiones de la plaza
plaza_width = width * 2
plaza_depth = depth * 2

# --- Crear el edificio ---

# Crear el cubo principal del edificio
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, building_height / 2))
building = bpy.context.object
building.scale = (width / 2, depth / 2, building_height / 2)
building.name = "Building"

# Crear la fachada de cristal
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, building_height / 2))
facade = bpy.context.object
facade.scale = (width / 2, depth / 2, 1)
facade.name = "Facade"
facade.material = bpy.data.materials.new(name="GlassMaterial")
facade.material.use_nodes = True
bsdf = facade.material.node_tree.nodes["Principled BSDF"]
bsdf.inputs["Transmission"].default_value = 1.0
bsdf.inputs["Roughness"].default_value = 0.0

# Crear columnas estructurales
column_radius = 0.5
column_height = building_height
num_columns_width = 3
num_columns_depth = 2
column_spacing_width = width / (num_columns_width + 1)
column_spacing_depth = depth / (num_columns_depth + 1)

for i in range(num_columns_width):
    for j in range(num_columns_depth):
        bpy.ops.mesh.primitive_cylinder_add(radius=column_radius, depth=column_height, enter_editmode=False, align='WORLD', location=(
            -width / 2 + (i + 0.5) * width / num_columns_width,
            -depth / 2 + (j + 0.5) * depth / num_columns_depth,
            column_height / 2
        ))
        column = bpy.context.object
        column.name = f"Column_{i}_{j}"

# --- Crear la plaza ---

# Crear el suelo de la plaza
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
plaza_floor = bpy.context.object
plaza_floor.scale = (plaza_width / 2, plaza_depth / 2, 1)
plaza_floor.name = "PlazaFloor"

# --- Ajustes finales ---

# Centrar el edificio en el origen
bpy.ops.object.origin_set(type='ORIGIN_WORLD', center='BOUNDS')

# --- Guardar el archivo ---
if "BLEND_OUT" in os.environ:
    filepath = os.environ["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=filepath)
else:
    print("No se especificó BLEND_OUT. No se guardó el archivo.")

import os
```
