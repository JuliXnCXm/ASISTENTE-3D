```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# --- Funciones de utilidad ---
def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color + (1,)  #RGBA
    return mat

# --- Materiales ---
material_wood = create_material("Wood", (0.6, 0.4, 0.2))
material_metal = create_material("Metal", (0.2, 0.2, 0.2))

# --- Estantería ---
shelf_width = 2.0
shelf_height = 0.2
shelf_depth = 0.4
num_shelves = 5
total_height = num_shelves * shelf_height

bpy.ops.mesh.primitive_cube_add(size=shelf_width, location=(0, 0, total_height / 2))
shelf = bpy.context.object
shelf.scale = (1, 1, shelf_height / shelf_width)
shelf.location = (0, 0, total_height / 2)
shelf.name = "Shelf"
shelf.data.materials.append(material_wood)

for i in range(1, num_shelves):
    bpy.ops.object.duplicate_move(OBJECT_OT_duplicate={"linked":False, "mode": 'TRANSLATION'}, TRANSFORM_OT_translate={"value": (0, 0, shelf_height)})
    new_shelf = bpy.context.object
    new_shelf.name = "Shelf." + str(i)
    new_shelf.data.materials.append(material_wood)

# --- Escritorio en forma de L ---
profile_curve_data = bpy.data.curves.new('profile_curve', 'CURVE')
profile_curve_data.dimensions = '3D'
profile_curve_data.resolution_u = 2
profile_curve_data.splines.new('BEZIER')
spline = profile_curve_data.splines[0]
spline.bezier_points.add(3)
spline.bezier_points[0].co = (0, 0, 0)
spline.bezier_points[1].co = (1.5, 0, 0)
spline.bezier_points[2].co = (1.5, 0.8, 0)
spline.bezier_points[3].co = (2.5, 0.8, 0)

profile_curve_object = bpy.data.objects.new('profile_curve_object', profile_curve_data)
bpy.context.collection.objects.link(profile_curve_object)

bpy.ops.object.convert(target='MESH')
desk_profile = bpy.context.object
desk_profile.name = "DeskProfile"

bpy.ops.object.editmode_toggle()
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_translate={"value":(0, 0, 0.8)})
bpy.ops.object.editmode_toggle()

desk = bpy.context.object
desk.name = "Desk"
desk.scale = (1, 1, 1)
desk.data.materials.append(material_wood)

# --- Silla ---
bpy.ops.mesh.primitive_cylinder_add(radius=0.5, depth=0.8, location=(1, 0, 0.4))
chair = bpy.context.object
chair.name = "Chair"
chair.data.materials.append(material_metal)

# --- Ajustes finales ---
bpy.context.scene.unit_settings.system_unit = 'METRIC'
bpy.context.scene.unit_settings.scale_length = 1.0

# Guardar el archivo .blend
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
