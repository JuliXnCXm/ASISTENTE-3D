```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de la sala de estar (en metros)
room_width = 6
room_depth = 8
room_height = 2.8

# Crear la sala de estar (un cubo)
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, room_height/2), scale=(room_width, room_depth, room_height))
room = bpy.context.object
room.name = "Room"

# Crear el sofá
sofa_width = 2.5
sofa_depth = 1.0
sofa_height = 0.7

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, -sofa_depth/2, sofa_height/2), scale=(sofa_width, 1, sofa_height))
sofa = bpy.context.object
sofa.name = "Sofa"

# Crear la alfombra
carpet_width = 3
carpet_depth = 2
carpet_thickness = 0.1

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, -carpet_depth/2, carpet_thickness/2), scale=(carpet_width, 1, carpet_thickness))
carpet = bpy.context.object
carpet.name = "Carpet"

# Crear el mueble de TV
tv_width = 3
tv_depth = 0.4
tv_height = 0.6

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, room_depth - tv_depth/2, tv_height/2), scale=(tv_width, 1, tv_height))
tv_unit = bpy.context.object
tv_unit.name = "TV_Unit"

# Crear el panel de listones de madera
wood_panel_width = room_width
wood_panel_height = 2
wood_panel_depth = 0.1
wood_panel_spacing = 0.1

bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(room_width/2, room_depth/2, wood_panel_height), scale=(wood_panel_width, 1, 1))
wood_panel = bpy.context.object
wood_panel.name = "Wood_Panel"

# Subdividir el panel para crear las listones
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.subdivide(number_cuts=int(wood_panel_width / (wood_panel_spacing + 0.1)))
bpy.ops.object.mode_set(mode='OBJECT')

# Separar las listones
bpy.ops.object.convert(target='MESH')
for i, obj in enumerate(bpy.context.selected_objects):
    obj.name = f"Wood_Strip_{i}"
    obj.location.z = wood_panel_height/2
    obj.scale.z = 1

# Guardar el archivo .blend
if 'BLEND_OUT' in os.environ:
    filepath = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=filepath)
```
