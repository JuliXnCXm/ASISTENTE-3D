```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# --- Funciones de utilidad ---
def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color + (1,)  #RGBA
    return mat

def create_cube(name, size, location, material=None):
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    cube = bpy.context.object
    cube.name = name
    if material:
        cube.data.materials.append(material)
    return cube

def create_plane(name, size_x, size_y, location, material=None):
    bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=location, scale=(1, 1, 1))
    plane = bpy.context.object
    plane.name = name
    plane.scale = (size_x, size_y, 1)
    if material:
        plane.data.materials.append(material)
    return plane

# --- Materiales ---
wall_material = create_material("WallMaterial", (0.8, 0.8, 0.8, 1))
floor_material = create_material("FloorMaterial", (0.9, 0.9, 0.9, 1))
bed_material = create_material("BedMaterial", (0.7, 0.3, 0.1, 1))
upholstered_headboard_material = create_material("UpholsteredHeadboardMaterial", (0.6, 0.5, 0.4, 1))
wardrobe_material = create_material("WardrobeMaterial", (0.5, 0.4, 0.3, 1))


# --- Dimensiones (en metros) ---
room_width = 6
room_length = 8
room_height = 3

bed_width = 2
bed_length = 2.2
bed_height = 0.5

headboard_height = 0.8

wardrobe_depth = 0.6
wardrobe_height = 2.4
wardrobe_width = 3

# --- Creación de la habitación ---
room_floor = create_plane("RoomFloor", room_width, room_length, (0, 0, 0), floor_material)
room_walls = []

# Paredes
wall_thickness = 0.1
wall1 = create_cube("Wall1", wall_thickness, (0, room_length/2, room_height/2), wall_material)
wall1.scale = (1, room_length, room_height)
wall2 = create_cube("Wall2", wall_thickness, (room_width/2, 0, room_height/2), wall_material)
wall2.scale = (room_width, 1, room_height)
wall3 = create_cube("Wall3", wall_thickness, (room_width, room_length/2, room_height/2), wall_material)
wall3.scale = (1, room_length, room_height)
wall4 = create_cube("Wall4", wall_thickness, (room_width/2, room_length, room_height/2), wall_material)
wall4.scale = (room_width, 1, room_height)

# --- Creación de la cama ---
bed_base = create_cube("BedBase", bed_width, (0, 0, bed_height), bed_material)
bed_base.scale = (1, 1, 1)

# Cabecero tapizado
headboard = create_cube("Headboard", bed_width, (0, -bed_length/2, headboard_height), upholstered_headboard_material)
headboard.scale = (1, 1, 1)

# --- Creación del muro separador y armarios ---
wardrobe_location_x = room_width / 2
wardrobe_location_y = room_length - wardrobe_depth

wardrobe = create_cube("Wardrobe", wardrobe_width, (wardrobe_location_x, wardrobe_location_y, wardrobe_height/2), wardrobe_material)
wardrobe.scale = (1, 1, 1)

# --- Opcional: Guardar el archivo ---

    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
