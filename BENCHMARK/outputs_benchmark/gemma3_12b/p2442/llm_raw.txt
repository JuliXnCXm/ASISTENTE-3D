```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de los muros en metros
wall_height = 2.5
wall_x_length = 4.0
wall_y_length = 3.0

# Crear el muro en el eje X
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, wall_y_length/2, wall_height/2), rotation=(0, 0, 0))
wall_x = bpy.context.object
wall_x.scale = (wall_x_length, 0.1, wall_height)
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
wall_x.name = "WallX"

# Crear el muro en el eje Y
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(wall_x_length/2, 0, wall_height/2), rotation=(0, 0, 0))
wall_y = bpy.context.object
wall_y.scale = (0.1, wall_y_length, wall_height)
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
wall_y.name = "WallY"

# Opcional: Guardar el archivo .blend
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
