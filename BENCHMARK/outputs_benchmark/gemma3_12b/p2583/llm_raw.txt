```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de la habitación en metros
room_width = 4.0
room_depth = 3.5
room_height = 2.5

# Dimensiones de la cama en metros
bed_width = 1.0
bed_depth = 2.0

# Dimensiones de la mesita de noche en metros
nightstand_width = 0.4
nightstand_depth = 0.4
nightstand_height = 0.5

# Crear paredes
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, room_height / 2), scale=(room_width / 2, room_depth / 2, room_height / 2))
room_wall_back = bpy.context.object
room_wall_back.name = "WallBack"

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, room_depth, room_height / 2), scale=(room_width / 2, room_height / 2, 0.01))
room_wall_right = bpy.context.object
room_wall_right.name = "WallRight"

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(room_width, 0, room_height / 2), scale=(room_width / 2, room_height / 2, 0.01))
room_wall_left = bpy.context.object
room_wall_left.name = "WallLeft"

bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, room_depth, room_height / 2), scale=(0.01, room_depth / 2, room_height / 2))
room_wall_front = bpy.context.object
room_wall_front.name = "WallFront"

# Crear suelo
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0), scale=(room_width / 2, room_depth / 2, 1))
floor = bpy.context.object
floor.name = "Floor"

# Crear cama
bed_x = -bed_depth / 2
bed_y = 0
bed_z = 0
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(bed_x, bed_y, bed_z), scale=(bed_width / 2, bed_depth / 2, 0.2))
bed = bpy.context.object
bed.name = "Bed"

# Crear mesita de noche
nightstand_x = bed_x + bed_depth
nightstand_y = -nightstand_depth / 2
nightstand_z = 0.25
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(nightstand_x, nightstand_y, nightstand_z), scale=(nightstand_width / 2, nightstand_depth / 2, nightstand_height / 2))
nightstand = bpy.context.object
nightstand.name = "Nightstand"

# Opcional: Guardar el archivo .blend
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
