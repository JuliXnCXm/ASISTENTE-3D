```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones en metros
pergola_width = 4.0
pergola_depth = 3.0
pergola_height = 2.5
post_width = 0.15
post_depth = 0.15
post_height = 2.5

# Crear postes
def create_post(x, y, z):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, z))
    post = bpy.context.object
    post.scale = (post_width / 2, post_depth / 2, post_height / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return post

post1 = create_post(pergola_width / 2 - post_width / 2, pergola_depth / 2 - post_depth / 2, post_height / 2)
post2 = create_post(pergola_width / 2 + post_width / 2, pergola_depth / 2 - post_depth / 2, post_height / 2)
post3 = create_post(pergola_width / 2 - post_width / 2, pergola_depth / 2 + post_depth / 2, post_height / 2)
post4 = create_post(pergola_width / 2 + post_width / 2, pergola_depth / 2 + post_depth / 2, post_height / 2)

# Crear vigas superiores
def create_beam(x, y, z, length, width, depth):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, z))
    beam = bpy.context.object
    beam.scale = (length / 2, width / 2, depth / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return beam

# Viga superior 1 (norte-sur)
beam1 = create_beam(pergola_width / 2, pergola_depth / 2, post_height, pergola_width, post_width, post_depth)

# Viga superior 2 (este-oeste)
beam2 = create_beam(pergola_width / 2, pergola_depth / 2, post_height, pergola_depth, post_width, post_depth)

# Crear diagonales
def create_diagonal(x1, y1, z1, x2, y2, z2, width, depth):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    diag = bpy.context.object
    length = math.sqrt((x2 - x1)**2 + (y2 - y1)**2 + (z2 - z1)**2)
    diag.scale = (length / 2, width / 2, depth / 2)
    
    # Rotar para alinear con la diagonal
    dx = x2 - x1
    dy = y2 - y1
    dz = z2 - z1
    angle_x = math.degrees(math.atan2(dy, dx))
    angle_z = math.degrees(math.atan2(dz, math.sqrt(dx**2 + dy**2)))
    
    diag.rotation_euler[0] = math.radians(angle_x)
    diag.rotation_euler[2] = math.radians(angle_z)
    
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return diag

# Diagonal 1
diag1 = create_diagonal(pergola_width / 2 - post_width / 2, pergola_depth / 2 - post_depth / 2, post_height, pergola_width / 2 + post_width / 2, pergola_depth / 2 - post_depth / 2, post_height)

# Diagonal 2
diag2 = create_diagonal(pergola_width / 2 - post_width / 2, pergola_depth / 2 + post_depth / 2, post_height, pergola_width / 2 + post_width / 2, pergola_depth / 2 + post_depth / 2, post_height)
```
