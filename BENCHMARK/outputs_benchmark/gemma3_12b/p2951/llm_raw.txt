```python
import bpy
import os

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones de la plaza
plaza_width = 12.0
plaza_depth = 10.0

# Dimensiones de los bancos
banco_largo = 2.0
banco_ancho = 0.6
banco_alto_asiento = 0.45
banco_alto_respaldo = 0.7
banco_separacion = 4.0

# Materiales
def crear_material(nombre, color):
    material = bpy.data.materials.new(name=nombre)
    material.use_nodes = True
    bsdf = material.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color + (1.0,)
    return material

material_suelo = crear_material("Suelo", (0.6, 0.4, 0.2))
material_madera = crear_material("Madera", (0.8, 0.6, 0.4))
material_hormigon = crear_material("Hormigon", (0.5, 0.5, 0.5))


# Crear suelo
bpy.ops.mesh.primitive_plane_add(size=plaza_width, enter_editmode=False, align='WORLD', location=(0, 0, 0))
suelo = bpy.context.object
suelo.name = "Suelo"
suelo.data.materials.append(material_suelo)

# Crear banco
def crear_banco(ubicacion):
    # Soportes de hormigón
    bpy.ops.mesh.primitive_cube_add(size=0.2, enter_editmode=False, align='WORLD', location=(ubicacion[0] + banco_largo/2 - 0.1, ubicacion[1] + banco_ancho/2, banco_alto_asiento/2))
    soporte1 = bpy.context.object
    soporte1.name = "Soporte1"
    soporte1.data.materials.append(material_hormigon)

    bpy.ops.mesh.primitive_cube_add(size=0.2, enter_editmode=False, align='WORLD', location=(ubicacion[0] + banco_largo/2 - 0.1, ubicacion[1] - banco_ancho/2, banco_alto_asiento/2))
    soporte2 = bpy.context.object
    soporte2.name = "Soporte2"
    soporte2.data.materials.append(material_hormigon)

    # Asiento de madera
    bpy.ops.mesh.primitive_plane_add(size=banco_largo, enter_editmode=False, align='WORLD', location=(ubicacion[0] + banco_largo/2, ubicacion[1] + banco_ancho/2, banco_alto_asiento))
    asiento = bpy.context.object
    asiento.name = "Asiento"
    asiento.data.materials.append(material_madera)

    # Respaldo de madera
    bpy.ops.mesh.primitive_plane_add(size=banco_largo/2, enter_editmode=False, align='WORLD', location=(ubicacion[0] + banco_largo/2, ubicacion[1] + banco_ancho/2, banco_alto_asiento + banco_alto_respaldo/2))
    respaldo = bpy.context.object
    respaldo.name = "Respaldo"
    respaldo.data.materials.append(material_madera)
    respaldo.rotation_euler[0] = -1.5708  # Rotar 90 grados en X

    # Agrupar los elementos del banco
    banco = bpy.data.objects.new("Banco", None)
    banco.parent = banco
    banco.matrix_world = banco.matrix_world
    soporte1.parent = banco
    soporte2.parent = banco
    asiento.parent = banco
    respaldo.parent = banco

    return banco

# Crear bancos enfrentados
banco1 = crear_banco((0, -plaza_depth/2 + banco_separacion/2, 0))
banco2 = crear_banco((plaza_width - banco_separacion, -plaza_depth/2, 0))

# Guardar la escena
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
