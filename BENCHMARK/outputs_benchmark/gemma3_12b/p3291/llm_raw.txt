```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones generales
building_width = 10  # metros
building_depth = 15  # metros
roof_height = 5  # metros
terrace_width = building_width
terrace_depth = building_depth
terrace_height = 0.1  # Grosor de la terraza

# Dimensiones de la pérgola
pergola_width = 6
pergola_depth = 8
pergola_height = 3
post_radius = 0.1
post_height = pergola_height + roof_height

# Dimensiones de las jardineras
planter_width = 0.3
planter_height = 0.5
planter_depth = 0.3

# Crear el edificio (solo la base para la terraza)
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, roof_height), scale=(building_width, building_depth, 1))
building = bpy.context.object
building.name = "Building"
building.data.name = "Building_Data"

# Crear la terraza de madera
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, roof_height), scale=(terrace_width, terrace_depth, 1))
terrace = bpy.context.object
terrace.name = "Terrace"
terrace.data.name = "Terrace_Data"
terrace.scale[2] = terrace_height

# Crear las jardineras
def create_planter(x, y, width, depth, height):
    bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(x, y, roof_height + height/2), scale=(width, depth, height))
    planter = bpy.context.object
    planter.name = "Planter"
    planter.data.name = "Planter_Data"
    return planter

# Jardineras perimetrales
planter_front = create_planter(0, terrace_depth/2, planter_width, terrace_depth, planter_height)
planter_back = create_planter(0, -terrace_depth/2, planter_width, terrace_depth, planter_height)
planter_left = create_planter(-terrace_width/2, 0, planter_width, terrace_depth, planter_height)
planter_right = create_planter(terrace_width/2, 0, planter_width, terrace_depth, planter_height)

# Crear la pérgola
def create_post(x, y, z, radius, height):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, enter_editmode=False, align='WORLD', location=(x, y, z), scale=(1, 1, 1))
    post = bpy.context.object
    post.name = "PergolaPost"
    post.data.name = "PergolaPost_Data"
    return post

# Crear postes de la pérgola
post_front_left = create_post(-pergola_width/2, -pergola_depth/2, roof_height + post_height/2, post_radius, post_height)
post_front_right = create_post(pergola_width/2, -pergola_depth/2, roof_height + post_height/2, post_radius, post_height)
post_back_left = create_post(-pergola_width/2, pergola_depth/2, roof_height + post_height/2, post_radius, post_height)
post_back_right = create_post(pergola_width/2, pergola_depth/2, roof_height + post_height/2, post_radius, post_height)

# Crear la estructura superior de la pérgola (simplificada como un cubo)
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, roof_height + pergola_height), scale=(pergola_width, pergola_depth, 1))
pergola_top = bpy.context.object
pergola_top.name = "PergolaTop"
pergola_top.data.name = "PergolaTop_Data"

# Crear vegetación (ejemplo: árboles simples)
def create_tree(x, y, z):
    bpy.ops.mesh.primitive_cylinder_add(radius=0.2, depth=1, enter_editmode=False, align='WORLD', location=(x, y, z), scale=(1, 1, 1))
    tree = bpy.context.object
    tree.name = "Tree"
    tree.data.name = "Tree_Data"
    return tree

# Colocar algunos árboles en las jardineras
tree1 = create_tree(-building_width/2 + 1, terrace_depth/2 - 0.5, roof_height + 1)
tree2 = create_tree(building_width/2 - 1, -terrace_depth/2 - 0.5, roof_height + 1)
tree3 = create_tree(-terrace_width/2 + 1, pergola_depth/2 - 0.5, roof_height + 1)
tree4 = create_tree(terrace_width/2 - 1, pergola_depth/2 - 0.5, roof_height + 1)

# Opcional: Guardar el archivo .blend
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
