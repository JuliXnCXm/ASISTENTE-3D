```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# --- Funciones de utilidad ---
def create_cube(size, location):
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    return bpy.context.object

def create_plane(size, location):
    bpy.ops.mesh.primitive_plane_add(size=size, location=location)
    return bpy.context.object

def set_material(obj, color, roughness=0.5):
    mat = bpy.data.materials.new(name="Material")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color + (1,)
    bsdf.inputs["Roughness"].default_value = roughness
    obj.data.materials.append(mat)

# --- Dimensiones y parámetros ---
wall_height = 2.8  # metros
wall_thickness = 0.2  # metros
floor_thickness = 0.1  # metros
tile_size = 0.3  # metros
house_length = 14  # metros
house_width = 8  # metros

# --- Creación de la casa en forma de 'L' ---

# Ala de día (sala de estar, comedor, cocina)
day_wing_length = house_length / 2
day_wing_width = house_width / 2

# Ala de noche (dormitorios, baño)
night_wing_length = house_length / 2
night_wing_width = house_width / 2

# --- Muros del ala de día ---
wall_day_1 = create_cube(day_wing_length, (0, day_wing_width / 2, wall_height / 2))
wall_day_2 = create_cube(day_wing_width, (day_wing_length / 2, 0, wall_height / 2))
wall_day_3 = create_cube(day_wing_length, (0, -day_wing_width / 2, wall_height / 2))
wall_day_4 = create_cube(day_wing_width, (-day_wing_length / 2, 0, wall_height / 2))

set_material(wall_day_1, (0.8, 0.8, 0.8))
set_material(wall_day_2, (0.8, 0.8, 0.8))
set_material(wall_day_3, (0.8, 0.8, 0.8))
set_material(wall_day_4, (0.8, 0.8, 0.8))

# --- Muros del ala de noche ---
wall_night_1 = create_cube(night_wing_length, (day_wing_length, night_wing_width / 2, wall_height / 2))
wall_night_2 = create_cube(night_wing_width, (day_wing_length + night_wing_length / 2, 0, wall_height / 2))
wall_night_3 = create_cube(night_wing_length, (day_wing_length, -night_wing_width / 2, wall_height / 2))
wall_night_4 = create_cube(night_wing_width, (day_wing_length + night_wing_length / 2, 0, wall_height / 2))

set_material(wall_night_1, (0.8, 0.8, 0.8))
set_material(wall_night_2, (0.8, 0.8, 0.8))
set_material(wall_night_3, (0.8, 0.8, 0.8))
set_material(wall_night_4, (0.8, 0.8, 0.8))

# --- Losa del patio central ---
patio_length = day_wing_length
patio_width = day_wing_width
patio = create_plane(patio_length, (0, 0, 0))
patio.scale = (patio_length / 2, patio_width / 2, 1)
set_material(patio, (0.7, 0.6, 0.4), roughness=0.2)

# --- Suelo de baldosas en el patio ---
def create_tile(x, y):
    tile = create_plane(1, (x, y, 0))
    tile.scale = (tile_size / 2, tile_size / 2, 1)
    return tile

for i in range(int(patio_length / tile_size)):
    for j in range(int(patio_width / tile_size)):
        create_tile(i * tile_size, j * tile_size)

# --- Mobiliario (representativo) ---
# Sala de estar
sofa = create_cube(2, (0, day_wing_width / 2 - 1, 0.5))
sofa.scale = (1, 0.8, 0.5)
set_material(sofa, (0.3, 0.2, 0.1))

# Cocina (isla)
island = create_cube(2, (day_wing_length / 2, day_wing_width / 2 - 1, 0.5))
island.scale = (1.5, 0.8, 0.5)
set_material(island, (0.2, 0.2, 0.2))

# Comedor (mesa)
table = create_cube(1.5, (day_wing_length / 2, -day_wing_width / 2 + 1, 0.5))
table.scale = (0.75, 0.75, 0.5)
set_material(table, (0.4, 0.4, 0.4))

# Dormitorio principal (cama)
bed = create_cube(1.8, (day_wing_length + 1, night_wing_width / 2 - 1, 0.5))
bed.scale = (0.9, 0.8, 0.5)
set_material(bed, (0.3, 0.1, 0.1))

# --- Ajustes finales ---
for obj in bpy.context.scene.objects:
    obj.select_set(False)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT está definida
if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
