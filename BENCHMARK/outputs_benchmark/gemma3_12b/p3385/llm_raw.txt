```python
import bpy
import math

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Dimensiones generales (en metros)
total_area = 55
apartment_width = 6  # Aproximadamente
apartment_depth = 9.17 # Aproximadamente (55 / 6)
wall_thickness = 0.15

# Funciones de utilidad
def create_wall(x1, y1, x2, y2, height, thickness):
    """Crea una pared entre dos puntos."""
    dx = x2 - x1
    dy = y2 - y1
    length = math.sqrt(dx**2 + dy**2)
    angle = math.atan2(dy, dx)

    bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
    wall = bpy.context.object
    wall.scale = (length / 2, thickness / 2, height / 2)
    wall.rotation_euler[2] = math.radians(math.degrees(angle))
    wall.location = ((x1 + x2) / 2, (y1 + y2) / 2, height / 2)
    return wall

# Crear paredes
# Pared exterior (L)
create_wall(0, 0, apartment_width, 0, 2.7, wall_thickness)
create_wall(apartment_width, 0, apartment_width, apartment_depth, 2.7, wall_thickness)
create_wall(apartment_width, apartment_depth, 0, apartment_depth, 2.7, wall_thickness)
create_wall(0, apartment_depth, 0, 0, 2.7, wall_thickness)

# Paredes internas
# Cocina
create_wall(apartment_width, 0, apartment_width, 0.8, 2.7, wall_thickness)
# Pasillo
create_wall(apartment_width, 0.8, apartment_width, 2.0, 2.7, wall_thickness)
# Baño
create_wall(apartment_width, 2.0, apartment_width, 3.5, 2.7, wall_thickness)
# Dormitorio
create_wall(apartment_width, 3.5, apartment_width, apartment_depth, 2.7, wall_thickness)

# Paredes de la sala-comedor
create_wall(0, 0, apartment_width, 0, 2.7, wall_thickness)
create_wall(0, 0, 0, apartment_depth, 2.7, wall_thickness)
create_wall(0, apartment_depth, apartment_width, apartment_depth, 2.7, wall_thickness)
create_wall(apartment_width, 0, apartment_width, 0, 2.7, wall_thickness)

# Crear suelo
bpy.ops.mesh.primitive_plane_add(size=apartment_width * 1.1, enter_editmode=False, align='WORLD', location=(apartment_width / 5, apartment_depth / 5, 0))
floor = bpy.context.object
floor.scale = (apartment_width / 2, apartment_depth / 2, 1)

# Crear techo
bpy.ops.mesh.primitive_plane_add(size=apartment_width * 1.1, enter_editmode=False, align='WORLD', location=(apartment_width / 5, apartment_depth / 5, 2.7))
ceiling = bpy.context.object
ceiling.scale = (apartment_width / 2, apartment_depth / 2, 1)
ceiling.location = (apartment_width / 5, apartment_depth / 5, 2.7)

# Ventanas (ejemplo, solo una en la sala y otra en el dormitorio)
# Ventana de la sala
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(apartment_width / 2, -apartment_depth / 2 + 0.1, 1.2))
window_living_room = bpy.context.object
window_living_room.scale = (1.5, 0.1, 0.1)
# Ventana del dormitorio
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(apartment_width / 2, apartment_depth / 2 + 0.1, 1.2))
window_bedroom = bpy.context.object
window_bedroom.scale = (1.5, 0.1, 0.1)

# Barra de la cocina (ejemplo)
bpy.ops.mesh.primitive_cube_add(size=0.2, enter_editmode=False, align='WORLD', location=(apartment_width - 0.1, 0.4, 0.9))
counter = bpy.context.object
counter.scale = (2, 0.2, 0.8)

# Guardar el archivo (si la variable de entorno BLEND_OUT está definida)
import os
if 'BLEND_OUT' in os.environ:
    filepath = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=filepath)
```
