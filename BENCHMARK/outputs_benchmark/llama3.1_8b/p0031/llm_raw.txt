import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la losa en metros
ancho = 6
alto = 4
espesor = 0.20

# Crea un plano rectangular con las dimensiones especificadas
bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
plano = bpy.context.active_object

# Establece la escala del plano para que coincida con las dimensiones de la losa
plano.scale = (ancho, alto, espesor)

# Renombra el objeto a "Losa de Piso"
bpy.context.active_object.name = 'Losa de Piso'

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
