```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de los peldaños
ancho_peldaño = 1.0
huella_peldaño = 0.3
contrahuella_peldaño = 0.18

# Crea el tramo de escalera
for i in range(15):
    # Crea un nuevo objeto para cada peldaño
    bpy.ops.mesh.primitive_cube_add(size=ancho_peldaño, location=(i * ancho_peldaño, 0, 0))
    
    # Obtiene el objeto actual (último creado)
    obj = bpy.context.active_object
    
    # Establece la altura del peldaño
    obj.scale[2] = contrahuella_peldaño
    
    # Mueve la huella del peldaño
    obj.location[1] = -huella_peldaño / 2
    
    # Duplica el objeto para crear la contrahuella
    bpy.ops.object.duplicate(obj=obj)
    
    # Obtiene la copia del objeto (último creado)
    dup_obj = bpy.context.active_object
    
    # Establece la altura de la contrahuella
    dup_obj.scale[2] = -contrahuella_peldaño
    
    # Mueve la contrahuella a su posición correcta
    dup_obj.location[1] = huella_peldaño / 2

# Si existe la variable de entorno BLEND_OUT, guarda el archivo .blend

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
