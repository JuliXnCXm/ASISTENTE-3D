```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades de la cama
alto_cama = 0.5  # Alto en metros
ancho_cama = 1.8  # Ancho en metros
profundidad_cama = 2.0  # Profundidad en metros

alto_madera = 0.01  # Alto de la madera en metros
anchura_madera = 0.05  # Ancho de la madera en metros

# Crea el cuerpo de la cama
bpy.ops.mesh.primitive_cube_add(size=ancho_cama, location=(0, 0, -alto_cama))
cama_body = bpy.context.active_object
cama_body.name = "Cama"

# Crea las patas de la cama
pata_alto = alto_madera
pata_ancho = anchura_madera

for i in range(4):
    pata = bpy.ops.mesh.primitive_cube_add(size=pata_ancho, location=(0, 0, -alto_cama + (i * (ancho_cama / 2))))
    pata = bpy.context.active_object
    if i == 0:
        pata.name = "Pata_Derecha_Frente"
    elif i == 1:
        pata.name = "Pata_Izquierda_Frente"
    elif i == 2:
        pata.name = "Pata_Derecha_Atrás"
    else:
        pata.name = "Pata_Izquierda_Atrás"

# Crea la estructura de madera
bpy.ops.mesh.primitive_cube_add(size=anchura_madera, location=(0, 0, -alto_cama + (ancho_cama / 2)))
madera_body = bpy.context.active_object
madera_body.name = "Estructura_Madera"

# Crea el colchón de tela blanca
colchon_alto = alto_cama
colchon_ancho = ancho_cama

bpy.ops.mesh.primitive_cube_add(size=colchon_ancho, location=(0, 0, -alto_cama))
colchon_body = bpy.context.active_object
colchon_body.name = "Colchón"

# Aplica materiales a los objetos
mat_madera = bpy.data.materials.new(name="Madera")
mat_madera.diffuse_color = (0.5, 0.2, 0.1)
cama_body.data.materials.append(mat_madera)

mat_tela = bpy.data.materials.new(name="Tela")
mat_tela.diffuse_color = (1, 1, 1)
colchon_body.data.materials.append(mat_tela)

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
