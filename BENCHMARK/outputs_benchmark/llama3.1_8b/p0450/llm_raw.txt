import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades de la geometría
length = 2.0  # metros
width = 1.5  # metros
height = 0.3  # metros
thickness = 0.05  # metros

# Crea el colchón blanco
bpy.ops.mesh.primitive_cube_add(size=2, location=(0, 0, height))
colchon = bpy.context.active_object
colchon.name = "Colchon"
colchon.data.materials.append(bpy.data.materials['Material.001'])  # Material blanco

# Crea la estructura de madera
bpy.ops.mesh.primitive_cube_add(size=2, location=(0, 0, height))
estructura = bpy.context.active_object
estructura.name = "Estructura"
estructura.data.materials.append(bpy.data.materials['Material.002'])  # Material madera

# Ajusta la escala de la estructura para que sea más delgada
estructura.scale[0] /= thickness / width
estructura.scale[1] /= thickness / length
estructura.scale[2] = height

# Combina el colchón y la estructura en un solo objeto
bpy.ops.object.join()

# Ajusta la posición de la cama para que esté centrada
cama = bpy.context.active_object
cama.location[0] -= length / 2
cama.location[1] -= width / 2

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
