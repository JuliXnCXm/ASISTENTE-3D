import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Establece las unidades a metros
bpy.context.scene.unit_settings.system = 'METRIC'
bpy.context.scene.unit_settings.length_unit = 'METERS'

# Crea el suelo de porcelanato
bpy.ops.mesh.primitive_plane_add(size=8, location=(0, 0, 0))
obj_suelo = bpy.context.active_object
obj_suelo.name = "Suelo"
obj_suelo.scale = (8, 6, 0.1)
mat_porcelanato = bpy.data.materials.new(name="Porcelanato")
mat_porcelanato.diffuse_color = (0.9, 0.7, 0.5, 1)
obj_suelo.data.materials.append(mat_porcelanato)

# Crea la zona de estar
bpy.ops.mesh.primitive_cube_add(size=2, location=(3, -2, 0))
obj_zona_estar = bpy.context.active_object
obj_zona_estar.name = "Zona de Estar"
obj_zona_estar.scale = (4, 6, 1)
mat_madera = bpy.data.materials.new(name="Madera")
mat_madera.diffuse_color = (0.8, 0.5, 0.2, 1)
obj_zona_estar.data.materials.append(mat_madera)

# Crea el sofá
bpy.ops.mesh.primitive_cube_add(size=1.5, location=(3.5, -2.5, 0))
obj_sofa = bpy.context.active_object
obj_sofa.name = "Sofá"
obj_sofa.scale = (4, 6, 1)
mat_tela = bpy.data.materials.new(name="Tela")
mat_tela.diffuse_color = (0.9, 0.7, 0.5, 1)
obj_sofa.data.materials.append(mat_tela)

# Crea la estantería
bpy.ops.mesh.primitive_cube_add(size=2, location=(3, -2, 1))
obj_estanteria = bpy.context.active_object
obj_estanteria.name = "Estantería"
obj_estanteria.scale = (4, 6, 1)
mat_madera = bpy.data.materials.new(name="Madera")
mat_madera.diffuse_color = (0.8, 0.5, 0.2, 1)
obj_estanteria.data.materials.append(mat_madera)

# Crea la zona de comedor
bpy.ops.mesh.primitive_cube_add(size=2, location=(-3, -2, 0))
obj_zona_comedor = bpy.context.active_object
obj_zona_comedor.name = "Zona de Comedor"

mat_madera = bpy.data.materials.new(name="Madera")

obj_zona_comedor.data.materials.append(mat_madera)

# Crea la mesa

obj_mesa = bpy.context.active_object
obj_mesa.name = "Mesa"
obj_mesa.scale = (4, 6, 1)
mat_madera = bpy.data.materials.new(name="Madera")
mat_madera.diffuse_color = (0.8, 0.5, 0.2, 1)
obj_mesa.data.materials.append(mat_madera)

# Crea las sillas
for i in range(4):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(-3.5 + i * 1.25, -2.75, 0))
    obj_silla = bpy.context.active_object
    obj_silla.name = f"Silla {i+1}"
    obj_silla.scale = (4, 6, 1)
    mat_madera = bpy.data.materials.new(name="Madera")
    mat_madera.diffuse_color = (0.8, 0.5, 0.2, 1)
    obj_silla.data.materials.append(mat_madera)

# Crea la luz de área
bpy.ops.mesh.primitive_circle_add(radius=1, location=(0, 0, 3))
obj_luz_area = bpy.context.active_object
obj_luz_area.name = "Luz de Área"
mat_luz = bpy.data.materials.new(name="Luz")
mat_luz.diffuse_color = (1, 1, 1, 1)
obj_luz_area.data.materials.append(mat_luz)
