```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del dormitorio
ancho = 4
alto = 3.5
profundidad = 2

# Crea el suelo de parquet
bpy.ops.mesh.primitive_plane_add(radius=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
obj_suelo = bpy.context.active_object
obj_suelo.scale = (ancho, alto, 0.01)
obj_suelo.location = (0, 0, -0.5)

# Crea las paredes del dormitorio
bpy.ops.mesh.primitive_plane_add(radius=1, enter_editmode=False, align='WORLD', location=(0, ancho/2, 0))
obj_pared_izquierda = bpy.context.active_object
obj_pared_izquierda.scale = (ancho, alto, profundidad)
obj_pared_izquierda.location = (-ancho/2, ancho/4, 0)

bpy.ops.mesh.primitive_plane_add(radius=1, enter_editmode=False, align='WORLD', location=(0, -ancho/2, 0))
obj_pared_derecha = bpy.context.active_object
obj_pared_derecha.scale = (ancho, alto, profundidad)
obj_pared_derecha.location = (ancho/2, -ancho/4, 0)

bpy.ops.mesh.primitive_plane_add(radius=1, enter_editmode=False, align='WORLD', location=(ancho/2, 0, 0))
obj_pared_frente = bpy.context.active_object
obj_pared_frente.scale = (alto, ancho, profundidad)
obj_pared_frente.location = (ancho/2, 0, 0)

bpy.ops.mesh.primitive_plane_add(radius=1, enter_editmode=False, align='WORLD', location=(-ancho/2, 0, 0))
obj_pared_atras = bpy.context.active_object
obj_pared_atras.scale = (alto, ancho, profundidad)
obj_pared_atras.location = (-ancho/2, 0, 0)

# Crea la cama doble con estructura de madera
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
obj_cama = bpy.context.active_object
obj_cama.scale = (2, 1.5, 0.5)
obj_cama.location = (ancho/4, ancho/2, -0.5)

# Crea el armario de dos puertas
bpy.ops.mesh.primitive_cube_add(size=1, enter_editmode=False, align='WORLD', location=(0, 0, 0))
obj_armario = bpy.context.active_object
obj_armario.scale = (ancho/2, alto, profundidad)
obj_armario.location = (-ancho/4, ancho/2, -0.5)

# Pinta las paredes de gris
bpy.data.materials.new(name='Gris')
mat_grey = bpy.data.materials['Gris']
mat_grey.diffuse_color = (0.7, 0.7, 0.7)
obj_pared_izquierda.data.materials.append(mat_grey)
obj_pared_derecha.data.materials.append(mat_grey)
obj_pared_frente.data.materials.append(mat_grey)
obj_pared_atras.data.materials.append(mat_grey)

# Guarda el archivo .blend si existe la variable de entorno BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
