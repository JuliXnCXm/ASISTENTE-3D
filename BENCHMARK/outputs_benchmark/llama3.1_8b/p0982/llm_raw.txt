```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades del sofá
sofa_name = "Sofa"
sofa_material = "Cuero"
sofa_size = (3, 1.5, 0.8)  # largo, ancho, alto en metros

# Crea el cuerpo del sofá
bpy.ops.mesh.primitive_cube_add(size=sofa_size[0], location=(0, -2, 0))
sofa_body = bpy.context.active_object
sofa_body.name = sofa_name + "_body"
sofa_body.scale = (sofa_size[0] / 1, sofa_size[1] / 1, sofa_size[2] / 1)

# Crea la parte de respaldo del sofá
bpy.ops.mesh.primitive_cube_add(size=sofa_size[0], location=(0, -2, 0.8))
sofa_back = bpy.context.active_object
sofa_back.name = sofa_name + "_back"
sofa_back.scale = (sofa_size[0] / 1, sofa_size[1] / 1, sofa_size[2] / 1)

# Crea la parte de asiento del sofá
bpy.ops.mesh.primitive_cube_add(size=sofa_size[0], location=(0, -2, 0.4))
sofa_seat = bpy.context.active_object
sofa_seat.name = sofa_name + "_seat"
sofa_seat.scale = (sofa_size[0] / 1, sofa_size[1] / 1, sofa_size[2] / 1)

# Combina las partes del sofá en un solo objeto
bpy.ops.object.join()

# Define las propiedades de la mesa
mesa_name = "Mesa"
mesa_material = "Nogal"
mesa_size = (1.5, 0.8, 0.4)  # largo, ancho, alto en metros

# Crea el cuerpo de la mesa
bpy.ops.mesh.primitive_cube_add(size=mesa_size[0], location=(2, -2, 0))
mesa_body = bpy.context.active_object
mesa_body.name = mesa_name + "_body"
mesa_body.scale = (mesa_size[0] / 1, mesa_size[1] / 1, mesa_size[2] / 1)

# Aplica el material al sofá y a la mesa
sofa_material_slot = sofa_body.material_slots.new(name=sofa_material)
sofa_material_slot.material = bpy.data.materials.get(sofa_material)

mesa_material_slot = mesa_body.material_slots.new(name=mesa_material)
mesa_material_slot.material = bpy.data.materials.get(mesa_material)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
