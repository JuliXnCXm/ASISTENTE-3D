```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la sala de estar en metros
sala_de_estar_dimensiones = (6, 4, 2.5)

# Crea el suelo de la sala de estar
bpy.ops.mesh.primitive_plane_add(radius=sala_de_estar_dimensiones[0] / 2, location=(0, 0, -sala_de_estar_dimensiones[2]))

# Define las dimensiones del sofá en metros
sofa_dimensiones = (3, 1.5, 0.8)

# Crea el sofá de tela gris
bpy.ops.mesh.primitive_cube_add(size=sofa_dimensiones[0], location=(0, -sala_de_estar_dimensiones[1] / 2 + sofa_dimensiones[1] / 2, 0))
sofa = bpy.context.active_object

# Aplica un material de color gris al sofá
mat = bpy.data.materials.new(name="Gris")
mat.diffuse_color = (0.5, 0.5, 0.5)
bpy.ops.object.select_all(action='DESELECT')
sofa.select_set(True)
bpy.context.collection.objects.link(sofa)
sofa.data.materials.append(mat)

# Define las dimensiones de la mesa de centro en metros
mesa_centro_dimensiones = (1.2, 0.8, 0.5)

# Crea la mesa de centro de madera de nogal
bpy.ops.mesh.primitive_cube_add(size=mesa_centro_dimensiones[0], location=(sofa_dimensiones[0] / 2 - mesa_centro_dimensiones[0] / 2, 0, sofa_dimensiones[1] + mesa_centro_dimensiones[1]))
mesa_centro = bpy.context.active_object

# Aplica un material de color madera al centro
mat = bpy.data.materials.new(name="Madera")
mat.diffuse_color = (0.8, 0.6, 0.4)
bpy.ops.object.select_all(action='DESELECT')
mesa_centro.select_set(True)
bpy.context.collection.objects.link(mesa_centro)
mesa_centro.data.materials.append(mat)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene['BLEND_OUT'])
```
