```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del dormitorio en metros
ancho = 4
alto = 3.5

# Crea el suelo
bpy.ops.mesh.primitive_plane_add(size=alto, location=(0, 0, -1))
bpy.context.active_object.name = "Suelo"
bpy.context.object.scale = (ancho, ancho, 1)

# Crea las paredes
bpy.ops.mesh.primitive_cube_add(size=2.5, location=(0, 0, 0))
bpy.context.active_object.name = "Pared Izquierda"
bpy.context.object.scale = (2.5, alto, 1)
bpy.context.object.location = (-ancho/2, 0, 0)

bpy.ops.mesh.primitive_cube_add(size=2.5, location=(0, 0, 0))
bpy.context.active_object.name = "Pared Derecha"
bpy.context.object.scale = (2.5, alto, 1)
bpy.context.object.location = (ancho/2, 0, 0)

bpy.ops.mesh.primitive_cube_add(size=4, location=(0, 0, 0))
bpy.context.active_object.name = "Pared Fondo"
bpy.context.object.scale = (ancho, alto, 1)
bpy.context.object.location = (0, 0, -alto)

bpy.ops.mesh.primitive_cube_add(size=3.5, location=(0, 0, 0))
bpy.context.active_object.name = "Pared Frente"
bpy.context.object.scale = (ancho, alto, 1)
bpy.context.object.location = (0, ancho/2, -alto)

# Pinta las paredes de gris
bpy.data.materials.new(name="Gris")
mat = bpy.data.materials["Gris"]
mat.diffuse_color = (0.5, 0.5, 0.5)
for obj in bpy.context.scene.objects:
    if "Pared" in obj.name:
        obj.active_material = mat

# Crea la cama doble
bpy.ops.mesh.primitive_cube_add(size=2, location=(0, 0, -1))
bpy.context.active_object.name = "Cama"
bpy.context.object.scale = (ancho/2, alto*0.7, 1)
bpy.context.object.location = (0, ancho/4, -alto)

# Crea la estructura de madera
bpy.ops.mesh.primitive_cube_add(size=0.5, location=(0, 0, -1))
bpy.context.active_object.name = "Estructura"
bpy.context.object.scale = (ancho/2*0.8, alto*0.7*0.8, 1)
bpy.context.object.location = (0, ancho/4, -alto-0.5)

# Guarda el archivo .blend si existe la variable de entorno BLEND_OUT
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene['BLEND_OUT'])
```
