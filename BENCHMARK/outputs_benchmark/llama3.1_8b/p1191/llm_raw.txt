```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la cama
ancho = 1.6
largo = 2.0
alto = 0.5

# Crea el marco de madera
bpy.ops.mesh.primitive_cube_add(size=alto, location=(0, 0, alto))
madera = bpy.context.active_object
madera.name = "Marco de Madera"
madera.scale = (ancho, largo, alto)
materiales = bpy.data.materials.new(name="Madera")
materiales.diffuse_color = (0.5, 0.2, 0.1)  # Color madera roble
materiales.specular_intensity = 0.5
materiales.use_transparency = True
materiales.alpha = 0.7
materiales.use_nodes = True
materiales.node_tree.nodes["Principled BSDF"].inputs['Base Color'].default_value = (0.5, 0.2, 0.1)
bpy.context.active_object.data.materials.append(materiales)

# Crea el colchón de textil blanco
bpy.ops.mesh.primitive_plane_add(size=ancho, location=(0, largo/2, alto + 0.05))
colchon = bpy.context.active_object
colchon.name = "Colchón"
colchon.scale = (ancho, ancho, alto)
materiales_colchon = bpy.data.materials.new(name="Textil Blanco")
materiales_colchon.diffuse_color = (1.0, 1.0, 1.0)  # Color blanco
materiales_colchon.specular_intensity = 0.5
bpy.context.active_object.data.materials.append(materiales_colchon)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene.BLEND_OUT)
```
