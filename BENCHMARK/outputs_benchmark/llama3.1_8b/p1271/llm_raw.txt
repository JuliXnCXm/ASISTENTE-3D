```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del terreno y la casa
terreno_ancho = 25
terreno_largo = 20
casa_ancho = 12
casa_largo = 9

# Crea el terreno de césped
bpy.ops.mesh.primitive_plane_add(size=terreno_ancho, location=(0, 0, 0))
bpy.context.active_object.name = "Terreno"
bpy.context.active_object.scale = (terreno_ancho / 2, terreno_largo / 2, 1)
bpy.ops.object.modifier_apply(modifier="Subdivision")

# Crea la casa
bpy.ops.mesh.primitive_cube_add(size=casa_ancho, location=(0, 0, 0))
casa = bpy.context.active_object
casa.name = "Casa"
casa.scale = (casa_ancho / 2, casa_largo / 2, 1)
bpy.ops.object.modifier_apply(modifier="Subdivision")

# Crea el tejado a dos aguas
bpy.ops.mesh.primitive_plane_add(size=casa_ancho, location=(0, 0, casa_largo))
tejado = bpy.context.active_object
tejado.name = "Tejado"
tejado.scale = (casa_ancho / 2, casa_largo / 2, 1)
bpy.ops.object.modifier_apply(modifier="Subdivision")
bpy.ops.mesh.primitive_plane_add(size=casa_ancho, location=(0, 0, -casa_largo))
tejado2 = bpy.context.active_object
tejado2.name = "Tejado2"
tejado2.scale = (casa_ancho / 2, casa_largo / 2, 1)
bpy.ops.object.modifier_apply(modifier="Subdivision")

# Combina los dos tejados para crear el tejado a dos aguas
bpy.context.view_layer.objects.active = tejado
bpy.ops.object.join()
bpy.context.view_layer.objects.active = tejado2
bpy.ops.object.join()

# Añade un árbol en el jardín frontal
bpy.ops.mesh.primitive_cylinder_add(radius=1, depth=casa_largo, location=(casa_ancho / 2, casa_largo / 2, 0))
arbol = bpy.context.active_object
arbol.name = "Arbol"
arbol.scale = (5, 10, 1)
bpy.ops.object.modifier_apply(modifier="Subdivision")

# Guarda el archivo .blend si se especificó la ruta de salida en la variable de entorno BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
