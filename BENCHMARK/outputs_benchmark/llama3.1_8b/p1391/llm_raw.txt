import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Establece las dimensiones de la parcela en metros
parcela_ancho = 5
parcela_largo = 5

# Crea un plano para representar la parcela
bpy.ops.mesh.primitive_plane_add(size=1, location=(0, 0, 0))
plano = bpy.context.active_object
plano.scale = (parcela_ancho, parcela_largo, 1)

# Crea el árbol de copa esférica
bpy.ops.mesh.primitive_uv_sphere_add(radius=2, enter_editmode=False, align='WORLD', location=(0, 0, 0))
arbol = bpy.context.active_object

# Ajusta la escala del árbol para que se adapte a las dimensiones de la parcela
arbol.scale = (1, 1, 2)

# Coloca el árbol en el centro de la parcela
arbol.location = (parcela_ancho / 2, parcela_largo / 2, 0)
