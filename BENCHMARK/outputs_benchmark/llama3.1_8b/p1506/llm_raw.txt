bpy.ops.wm.read_factory_settings(use_empty=True)

objeto = bpy.context.active_object

# Establece las dimensiones del plano en metros
objeto.scale = (ancho, alto, 0.01)

# Renombra el objeto para que sea más descriptivo
objeto.name = 'Terreno'
