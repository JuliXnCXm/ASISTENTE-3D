import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del muro y la puerta
largo_muro = 8
alto_muro = 2.2
ancho_puerta = 1

# Crea el muro
bpy.ops.mesh.primitive_cube_add(size=alto_muro, location=(0, 0, alto_muro/2))
muro = bpy.context.active_object
muro.scale((largo_muro, alto_muro, 1))

# Crea la puerta
bpy.ops.mesh.primitive_cube_add(size=alto_muro, location=(-largo_muro/2 + ancho_puerta/2, 0, alto_muro/2))
puerta = bpy.context.active_object
puerta.scale((ancho_puerta, alto_muro, 1))

# Mueve la puerta al centro del muro
puerta.location = (largo_muro/2 - ancho_puerta/2, 0, alto_muro/2)

# Dibuja una línea para separar el muro y la puerta
bpy.ops.mesh.primitive_cube_add(size=alto_muro, location=(-largo_muro/2 + ancho_puerta/2 - 0.01, 0, alto_muro/2))
separador = bpy.context.active_object
separador.scale((0.02, alto_muro, 1))
