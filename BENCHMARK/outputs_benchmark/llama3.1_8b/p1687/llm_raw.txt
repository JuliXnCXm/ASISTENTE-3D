```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del tramo de acera y la calzada
acera_longitud = 10
acera_ancho = 2
calzada_ancho = 4

# Crea el tramo de acera
bpy.ops.mesh.primitive_rectangle_add(size=acera_ancho, depth=acera_longitud)
acera = bpy.context.active_object
acera.location = mathutils.Vector((0, 0, 0))
acera.scale = (acera_longitud, acera_ancho, 1)

# Crea la calzada de asfalto
bpy.ops.mesh.primitive_rectangle_add(size=calzada_ancho, depth=acera_longitud)
calzada = bpy.context.active_object
calzada.location = mathutils.Vector((0, -acera_ancho/2 - calzada_ancho/2, 0))
calzada.scale = (acera_longitud, calzada_ancho, 1)

# Crea la baranda metálica
bpy.ops.mesh.primitive_cube_add(size=1)
baranda = bpy.context.active_object
baranda.location = mathutils.Vector((0, acera_ancho/2 + calzada_ancho/2, 1))
baranda.scale = (acera_longitud, 0.5, 0.5)

# Ajusta la altura de la baranda para que esté a 1 metro del suelo
baranda.location.z += 1

# Si existe la variable de entorno BLEND_OUT, guarda el archivo .blend
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
