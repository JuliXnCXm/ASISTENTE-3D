```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del muro y la puerta
longitud_muro = 10
alto_muro = 2.5
ancho_puerta = 1.5

# Crea el muro
bpy.ops.mesh.primitive_cube_add(size=alto_muro, location=(0, 0, alto_muro/2))
muro = bpy.context.active_object
muro.scale((longitud_muro, alto_muro, 1))

# Crea la puerta
bpy.ops.mesh.primitive_cube_add(size=alto_muro, location=(-longitud_muro/2 + ancho_puerta/2, 0, alto_muro/2))
puerta = bpy.context.active_object
puerta.scale((ancho_puerta, alto_muro, 1))

# Dibuja la textura del ladrillo rojo en el muro
muro.data.materials.append(bpy.data.materials.new('LadrilloRojo'))
muro.data.materials['LadrilloRojo'].diffuse_color = (0.8, 0.2, 0.2)

# Dibuja la textura de la madera en la puerta
puerta.data.materials.append(bpy.data.materials.new('Madera'))
puerta.data.materials['Madera'].diffuse_color = (1, 0.5, 0.2)

# Guarda el archivo .blend si se especificó la variable BLEND_OUT
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene.BLEND_OUT)
```
