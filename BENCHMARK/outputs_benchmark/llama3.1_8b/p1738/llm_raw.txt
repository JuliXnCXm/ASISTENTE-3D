```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades del árbol
radio_copa = 2  # en metros
alto_copa = 4   # en metros
alto_tronco = 2.5  # en metros

# Crea la copa del árbol
bpy.ops.mesh.primitive_circle_add(vertices=32, radius=radio_copa)
cabra_copa = bpy.context.active_object
cabra_copa.location = (0, 0, alto_tronco + alto_copa)

# Duplica y gira las caras para crear la copa completa
for i in range(16):
    cabra_copa.select_set(True)
    bpy.ops.object.duplicate_move()
    bpy.context.active_object.rotation_euler = mathutils.Euler((0, 0, (i * math.pi) / 8), 'Z')
    bpy.ops.object.join()

# Crea el tronco del árbol
bpy.ops.mesh.primitive_cylinder_add(radius=radio_copa/2, depth=alto_tronco)
tronco = bpy.context.active_object

# Ubica el tronco en su lugar correcto
tronco.location = (0, 0, alto_tronco)

# Si existe la variable de entorno BLEND_OUT, guarda el archivo .blend
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
