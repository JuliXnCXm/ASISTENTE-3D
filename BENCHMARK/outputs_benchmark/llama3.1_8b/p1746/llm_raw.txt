```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Establece las unidades a metros
bpy.context.scene.unit_settings.system = 'METRIC'
bpy.context.scene.unit_settings.length_unit = 'METERS'

# Crea la base de asfalto
base_asfalto = bpy.data.objects.new('BaseAsfalto', None)
base_asfalto.location = (0, 0, 0)
base_asfalto.scale = (10, 10, 0.1)  # 10x10 metros
bpy.context.collection.objects.link(base_asfalto)

# Crea la estructura de columnas
columna = bpy.data.objects.new('Columna', None)
columna.location = (5, 0, 0)
columna.scale = (0.2, 0.2, 10)  # 20x20 cm x 1m
bpy.context.collection.objects.link(columna)

# Crea la fachada de estuco
fachada_estuco = bpy.data.objects.new('FachadaEstuco', None)
fachada_estuco.location = (0, 0, 5)
fachada_estuco.scale = (10, 10, 2)  # 10x10 metros x 2m
bpy.context.collection.objects.link(fachada_estuco)

# Crea las ventanas en la fachada principal
ventana = bpy.data.objects.new('Ventana', None)
ventana.location = (5, 0, 5)
ventana.scale = (1, 2, 2)  # 1x2 metros x 2m
bpy.context.collection.objects.link(ventana)

# Crea las plantas del edificio
for i in range(5):
    planta = bpy.data.objects.new(f'Planta{i+1}', None)
    planta.location = (0, 0, 10 + i * 2)  # 10 metros de altura entre plantas
    planta.scale = (10, 10, 2)  # 10x10 metros x 2m
    bpy.context.collection.objects.link(planta)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
