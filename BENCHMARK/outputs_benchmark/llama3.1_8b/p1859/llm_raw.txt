```python
import bpy
import mathutils

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones del objeto
 metros = 1

# Crear el suelo
bpy.ops.mesh.primitive_plane_add(location=(0, 0, -metros))

# Crear la estantería
estanteria = bpy.data.objects['Plane']
estanteria.name = 'Estanteria'
estanteria.location = (0, metros * 2, 0)
estanteria.scale = (metros * 5, metros * 1, metros)

# Crear la silla
silla = bpy.ops.mesh.primitive_cube_add(location=(0, -metros * 3, 0))
silla.name = 'Silla'
silla.scale = (metros, metros, metros)

# Crear el escritorio
perfil = bpy.data.objects['Cube']
perfil.name = 'Perfil'
perfil.location = (0, 0, metros)
perfil.scale = (metros, metros * 1, metros)

bpy.ops.mesh.primitive_cube_add(location=(0, 0, metros))
escritorio = bpy.context.active_object
escritorio.name = 'Escritorio'

# Extruir el perfil para crear la forma L del escritorio
extrusion_settings = bpy.context.scene.tool_settings.extrapolate
bpy.context.scene.tool_settings.extrapolate = True

bpy.ops.mesh.extrude_individually(
    mesh=bpy.data.objects['Perfil'].data,
    type='REGION',
    use_normal=True,
    mirror=False,
    use_dissolve=False,
    use_decrease_edge=False
)

# Restaurar la configuración de extrusión
bpy.context.scene.tool_settings.extrapolate = extrusion_settings

# Posicionar el escritorio en su lugar
escritorio.location = (0, metros * 2, metros)
escritorio.scale = (metros * 5, metros * 1.5, metros)

# Guardar la escena si se especifica una ruta de salida
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_as_mainfile(filepath=bpy.context.scene['BLEND_OUT'])
```
