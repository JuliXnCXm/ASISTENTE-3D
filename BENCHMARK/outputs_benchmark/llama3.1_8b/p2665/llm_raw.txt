```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones y proporciones de los peldaños
huella = 0.28
contrahuella = 0.18
ancho_tramo = 1.0
alto_peldaño = 0.2

# Crea un objeto para el tramo de escalera
tramo = bpy.data.objects.new("Tramo", None)
bpy.context.collection.objects.link(tramo)

# Define la posición y orientación del tramo
tramo.location = mathutils.Vector((0, 0, 0))
tramo.rotation_euler = mathutils.Euler((0, 0, 0), 'XYZ')

# Crea los peldaños
for i in range(12):
    # Calcula la posición y orientación del peldaño actual
    x = (i - 6) * ancho_tramo / 2
    z = i * alto_peldaño
    
    # Crea el objeto para el peldaño actual
    peldaño = bpy.data.objects.new(f"Peldaño {i+1}", None)
    bpy.context.collection.objects.link(peldaño)
    
    # Define la posición y orientación del peldaño
    peldaño.location = mathutils.Vector((x, 0, z))
    peldaño.rotation_euler = mathutils.Euler((0, 0, 0), 'XYZ')
    
    # Agrega el objeto al tramo
    bpy.ops.object.select_all(action='DESELECT')
    peldaño.select_set(True)
    bpy.context.collection.objects.link(peldaño)
    
    # Crea la huella y contrahuella del peldaño actual
    if i < 11:
        # Huella
        huella_obj = bpy.data.objects.new(f"Huella {i+1}", None)
        bpy.context.collection.objects.link(huella_obj)
        
        huella_obj.location = mathutils.Vector((x + ancho_tramo / 2, -huella / 2, z))
        huella_obj.rotation_euler = mathutils.Euler((0, 0, 0), 'XYZ')
        
        # Contrahuella
        contrahuella_obj = bpy.data.objects.new(f"Contrahuella {i+1}", None)
        bpy.context.collection.objects.link(contrahuella_obj)
        
        contrahuella_obj.location = mathutils.Vector((x + ancho_tramo / 2, huella / 2, z))
        contrahuella_obj.rotation_euler = mathutils.Euler((0, 0, 0), 'XYZ')

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if 'BLEND_OUT' in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene['BLEND_OUT'])
```
