```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la pérgola
ancho = 4
alto = 3
altura = 2.5

# Define las dimensiones de los postes
ancho_poste = 0.15
alto_poste = 0.15

# Crea los postes
for i in range(4):
    # Calcula la posición del poste
    x = (i % 2) * ancho / 2 - ancho / 4
    y = (i // 2) * alto / 2 - alto / 4
    
    # Crea el poste
    bpy.ops.mesh.primitive_cube_add(location=(x, y, 0), size=alto_poste)
    objeto = bpy.context.active_object
    objeto.name = f"Poste {i+1}"
    
    # Rotula la posición del poste
    objeto.location = (x, y, altura)

# Crea el marco de la pérgola
bpy.ops.mesh.primitive_cube_add(location=(0, 0, altura), size=ancho)
objeto = bpy.context.active_object
objeto.name = "Marco"

# Rotula la posición del marco
objeto.location = (0, 0, altura)

# Crea los lados de la pérgola
for i in range(4):
    # Calcula la posición del lado
    x = (i % 2) * ancho / 2 - ancho / 4
    y = (i // 2) * alto / 2 - alto / 4
    
    # Crea el lado
    bpy.ops.mesh.primitive_cube_add(location=(x, y, altura), size=ancho)
    objeto = bpy.context.active_object
    objeto.name = f"Lado {i+1}"
    
    # Rotula la posición del lado
    objeto.location = (x, y, altura)
```
