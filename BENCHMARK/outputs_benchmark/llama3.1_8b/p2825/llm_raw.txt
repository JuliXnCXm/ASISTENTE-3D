import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades del bolardo
diameter = 0.2  # Diámetro en metros
height = 0.9    # Altura en metros

# Crea un cilindro a partir de una esfera
bpy.ops.mesh.primitive_cylinder_add(radius=diameter/2, depth=height)

# Renombra el objeto para que sea más descriptivo
bpy.context.active_object.name = "Bolardo"

# Establece la escala del objeto en metros
bpy.context.object.scale[0] = diameter / 2
bpy.context.object.scale[1] = diameter / 2
bpy.context.object.scale[2] = height

# Mueve el objeto al origen
bpy.context.object.location = mathutils.Vector((0, 0, 0))

    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
