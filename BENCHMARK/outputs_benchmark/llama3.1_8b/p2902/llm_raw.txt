```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las propiedades del bolardo
alto = 0.8
diametro = 0.3

# Crea el cilindro en el origen
cylinder = bpy.data.objects.new("Bolardo", bpy.data.meshes.new("Cilindro"))

# Define la geometría del cilindro
verts = [
    mathutils.Vector((0, alto / 2, 0)),
    mathutils.Vector((diametro / 2, alto / 2, 0)),
    mathutils.Vector((diametro / 2, -alto / 2, 0)),
    mathutils.Vector((0, -alto / 2, 0))
]

edges = [
    (0, 1),
    (1, 2),
    (2, 3),
    (3, 0)
]

faces = [
    (0, 1, 2, 3)
]

# Crea la geometría del cilindro
cylinder.data.from_pydata(verts, edges, faces)
cylinder.data.update()

# Coloca el cilindro en la escena
bpy.context.collection.objects.link(cylinder)

# Si existe la variable de entorno BLEND_OUT, guarda el archivo .blend
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
