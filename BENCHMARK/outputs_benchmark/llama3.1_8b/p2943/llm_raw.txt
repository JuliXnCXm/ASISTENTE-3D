```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la base de la cubierta
base_length = 12
base_width = 8

# Define la altura de la cumbrera y el espesor de la cubierta
cumbrera_height = 3
cubierta_thick = 0.25

# Crea un nuevo objeto para la base de la cubierta
bpy.ops.mesh.primitive_cube_add(size=base_width, location=(0, 0, 0))
base_mesh = bpy.context.active_object
base_mesh.name = "Base"

# Duplica el objeto y lo ajusta a las dimensiones de la base de la cubierta
bpy.ops.object.duplicate()
bpy.context.active_object.scale((base_length / base_width, 1, 1))

# Crea un nuevo objeto para la cubierta
bpy.ops.mesh.primitive_cube_add(size=cumbrera_height + cubierta_thick, location=(0, 0, cumbrera_height))
cubierta_mesh = bpy.context.active_object
cubierta_mesh.name = "Cubierta"

# Duplica el objeto y lo ajusta a las dimensiones de la base de la cubierta
bpy.ops.object.duplicate()
bpy.context.active_object.scale((base_length / base_width, 1, 1))

# Ajusta la posición de la cubierta para que coincida con la cumbrera
cubierta_mesh.location = (0, 0, cumbrera_height + cubierta_thick)

# Combina los objetos en un solo objeto
bpy.ops.object.join()

# Guarda el archivo .blend si se especificó la ruta de salida
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
