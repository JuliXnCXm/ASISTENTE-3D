```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones de la vivienda en metros
ancho = 10
profundo = 5
alto = 3

# Crea el suelo
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, -alto))
bpy.context.active_object.name = "Suelo"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea las paredes laterales
bpy.ops.mesh.primitive_cube_add(size=ancho, location=(-profundo/2, 0, 0))
bpy.context.active_object.name = "Pared Izquierda"
bpy.ops.object.modifier_apply(modifier="Displace")
bpy.ops.mesh.primitive_cube_add(size=ancho, location=(profundo/2, 0, 0))
bpy.context.active_object.name = "Pared Derecha"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea la pared trasera
bpy.ops.mesh.primitive_cube_add(size=profundo, location=(0, -ancho/2, 0))
bpy.context.active_object.name = "Pared Trasera"
bpy.ops.object.modifier_apply(modifier="Displace")
bpy.ops.mesh.primitive_cube_add(size=profundo, location=(0, ancho/2, 0))
bpy.context.active_object.name = "Pared Frente"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea el techo
bpy.ops.mesh.primitive_cube_add(size=ancho, location=(0, 0, alto))
bpy.context.active_object.name = "Techo"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea la sala-comedor y cocina
bpy.ops.mesh.primitive_cube_add(size=4, location=(-profundo/2 + 1.5, -ancho/2 + 1.5, 0))
bpy.context.active_object.name = "Sala-Cocina"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea el dormitorio principal
bpy.ops.mesh.primitive_cube_add(size=3, location=(-profundo/2 + 4, -ancho/2 + 1.5, 0))
bpy.context.active_object.name = "Dormitorio"
bpy.ops.object.modifier_apply(modifier="Displace")

# Crea el baño completo
bpy.ops.mesh.primitive_cube_add(size=3, location=(-profundo/2 + 7, -ancho/2 + 1.5, 0))
bpy.context.active_object.name = "Baño"
bpy.ops.object.modifier_apply(modifier="Displace")

# Guarda el archivo .blend si existe la variable de entorno BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
