```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del acceso monumental
ancho = 10  # metros
alto = 5  # metros
profundo = 20  # metros

# Crea el plano de base para la escalinata y rampas
base = bpy.data.objects.new("Base", bpy.data.meshes.new("Base"))
base.location = (0, 0, 0)
bpy.context.collection.objects.link(base)

# Crea la geometría de la escalinata
escalera = bpy.data.objects.new("Escalera", bpy.data.meshes.new("Escalera"))
escalera.location = (0, -alto/2, 0)
escalera.scale = (ancho, alto, profundo)
bpy.context.collection.objects.link(escalera)

# Crea las rampas de accesibilidad
rampa_derecha = bpy.data.objects.new("Rampa Derecha", bpy.data.meshes.new("Rampa Derecha"))
rampa_derecha.location = (-ancho/2, -alto/2, 0)
rampa_derecha.scale = (ancho, alto, profundo)
bpy.context.collection.objects.link(rampa_derecha)

rampa_izquierda = bpy.data.objects.new("Rampa Izquierda", bpy.data.meshes.new("Rampa Izquierda"))
rampa_izquierda.location = (ancho/2, -alto/2, 0)
rampa_izquierda.scale = (ancho, alto, profundo)
bpy.context.collection.objects.link(rampa_izquierda)

# Crea las jardineras de hormigón
jardinera_derecha = bpy.data.objects.new("Jardinera Derecha", bpy.data.meshes.new("Jardinera Derecha"))
jardinera_derecha.location = (-ancho/2, alto/2, 0)
jardinera_derecha.scale = (ancho, alto, profundo)
bpy.context.collection.objects.link(jardinera_derecha)

jardinera_izquierda = bpy.data.objects.new("Jardinera Izquierda", bpy.data.meshes.new("Jardinera Izquierda"))
jardinera_izquierda.location = (ancho/2, alto/2, 0)
jardinera_izquierda.scale = (ancho, alto, profundo)
bpy.context.collection.objects.link(jardinera_izquierda)

# Agrega iluminación de piso
lampara_piso = bpy.data.objects.new("Lampara Piso", bpy.data.lights.new("Lampara Piso", 'SPOT'))
lampara_piso.location = (0, 0, -alto/2)
bpy.context.collection.objects.link(lampara_piso)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if "BLEND_OUT" in bpy.context.scene:
    bpy.ops.wm.save_mainfile(filepath=bpy.context.scene["BLEND_OUT"])
```
