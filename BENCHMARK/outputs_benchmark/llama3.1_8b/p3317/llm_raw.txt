import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Establece las unidades en metros
bpy.context.scene.unit_settings.system = 'METRIC'
bpy.context.scene.unit_settings.length_unit = 'METERS'

# Crea el pavimento de adoquines de hormigón
pavimento = bpy.data.objects.new('Pavimento', None)
pavimento.location = mathutils.Vector((0, 0, 0))
pavimento.scale = (10, 1, 0.01)  # Tamaño del adoquín

# Crea un objeto de tipo plano para el pavimento
pavimento_data = bpy.data.meshes.new('Pavimento')
pavimento_mesh = bpy.data.objects.new('Pavimento', pavimento_data)
bpy.context.collection.objects.link(pavimento_mesh)

# Crea una serie de objetos para representar los adoquines
for i in range(10):
    for j in range(10):
        # Crea un objeto de tipo plano para el adoquín
        adoquin = bpy.data.meshes.new('Adoquin')
        adoquin_data = bpy.data.objects.new(f'Adoquin_{i}_{j}', adoquin)
        
        # Establece la posición del adoquín
        adoquin.location = mathutils.Vector((i * 1.2, j * 1.2, 0))
        
        # Crea un plano para el adoquín
        bpy.ops.mesh.primitive_plane_add(size=1, enter_editmode=False, align='WORLD', location=(i * 1.2, j * 1.2, 0), rotation=(0, 0, 0))
        plane = bpy.context.active_object
        
        # Convierte el plano en un objeto de tipo mesh
        plane_data = plane.data
        
        # Asigna la geometría del adoquín al objeto
        plane_data.name = 'Adoquin'
        
        # Agrega el objeto a la escena
        bpy.context.collection.objects.link(plane)
        
        # Elimina el objeto original
        bpy.ops.object.delete(use_global=False)

# Crea las farolas altas y delgadas para la iluminación nocturna
farola = bpy.data.objects.new('Farola', None)
farola.location = mathutils.Vector((0, 0, 2))
farola.scale = (1, 0.5, 3)  # Tamaño de la farola

# Crea un objeto de tipo cilindro para la farola
farola_data = bpy.data.meshes.new('Farola')
farola_mesh = bpy.data.objects.new('Farola', farola_data)
bpy.context.collection.objects.link(farola_mesh)

# Crea una serie de objetos para representar las farolas
for i in range(5):
    for j in range(5):
        # Crea un objeto de tipo cilindro para la farola
        farola_cilindro = bpy.data.meshes.new('Farola_Cilindro')
        farola_cilindro_data = bpy.data.objects.new(f'Farola_{i}_{j}', farola_cilindro)
        
        # Establece la posición de la farola
        farola_cilindro.location = mathutils.Vector((i * 2.5, j * 2.5, 2))
        
        # Crea un cilindro para la farola
        bpy.ops.mesh.primitive_cylinder_add(radius=0.25, depth=3, enter_editmode=False, align='WORLD', location=(i * 2.5, j * 2.5, 2), rotation=(0, 0, 0))
        cylinder = bpy.context.active_object
        
        # Convierte el cilindro en un objeto de tipo mesh

        bpy.ops.object.delete(use_global=False)

arbol = bpy.data.objects.new('Arbol', None)
arbol.location = mathutils.Vector((0, 0, 2))
arbol.scale = (1, 1.5, 3)  # Tamaño del árbol
