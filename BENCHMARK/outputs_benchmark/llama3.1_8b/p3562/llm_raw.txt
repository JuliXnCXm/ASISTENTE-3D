```python
import bpy
import mathutils

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Establece las unidades de medida a metros
bpy.context.scene.unit_settings.system = 'METRIC'
bpy.context.scene.unit_settings.length_unit = 'METERS'

# Crea el pavimento de adoquines de hormigón
pavimento = bpy.data.objects.new('Pavimento', None)
pavimento.location = (0, 0, 0)

# Define la forma del adoquín
adoquin = bpy.data.curves.new('Adoquin', 'MESH')
adoquin.dimensions = '2x1x0.05'
adoquin.shape_keys.active = 'Copy'

# Crea una serie de adoquines para formar el pavimento
for x in range(-10, 11):
    for y in range(-10, 11):
        if (x + y) % 2 == 0:
            nuevo_adoquin = bpy.data.objects.new(f'Adoquin_{x}_{y}', adoquin)
            nuevo_adoquin.location = (x * 1.5, y * 1.5, -0.05)
            pavimento.modifiers['Array'].offset = (x * 1.5, y * 1.5, 0)
            bpy.context.collection.objects.link(nuevo_adoquin)

# Crea las bancas de diseño contemporáneo
banca = bpy.data.objects.new('Banca', None)
banca.location = (-3, -3, 0)
banca.scale = (2, 1.5, 1)

# Duplica la banca para crear varias
for i in range(4):
    nueva_banca = bpy.data.objects.new(f'Nueva_Banca_{i}', banca)
    nueva_banca.location = (-3 + i * 3, -3, 0)
    bpy.context.collection.objects.link(nueva_banca)

# Crea las farolas altas para la iluminación nocturna
farola = bpy.data.objects.new('Farola', None)
farola.location = (5, 0, 2)
farola.scale = (1.5, 1.5, 3)

# Duplica la farola para crear varias
for i in range(4):
    nueva_farola = bpy.data.objects.new(f'Nueva_Farola_{i}', farola)
    nueva_farola.location = (5 + i * 3, 0, 2)
    bpy.context.collection.objects.link(nueva_farola)

# Crea el árbol central plantado en un alcorque elevado
arbol = bpy.data.objects.new('Arbol', None)
arbol.location = (0, 0, -1.5)
arbol.scale = (3, 2, 6)

alcorque = bpy.data.objects.new('Alcorque', None)
alcorque.location = (0, 0, -1.8)
alcorque.scale = (2, 2, 1)

# Duplica el árbol y el alcorque para crear varios
for i in range(4):
    nuevo_arbol = bpy.data.objects.new(f'Nuevo_Arbol_{i}', arbol)
    nuevo_alcorque = bpy.data.objects.new(f'Nuevo_Alcorque_{i}', alcorque)
    nuevo_arbol.location = (0, 0, -1.5 + i * 3)
    nuevo_alcorque.location = (0, 0, -1.8 + i * 3)
    bpy.context.collection.objects.link(nuevo_arbol)
    bpy.context.collection.objects.link(nuevo_alcorque)

# Guarda el archivo .blend si se especificó la variable de entorno BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_mainfile(filepath=os.environ['BLEND_OUT'])
```
