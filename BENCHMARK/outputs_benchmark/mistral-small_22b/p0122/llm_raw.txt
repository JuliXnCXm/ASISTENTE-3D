import bpy
import math

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Parámetros de la escalera
altura_total = 3.0
diametro = 1.8
numero_escalones = 16
radio_interno = diametro / 2 - 0.1
radio_externo = radio_interno + 0.2
altura_escalon = altura_total / numero_escalones

# Crear el poste central
bpy.ops.mesh.primitive_cylinder_add(radius=0.1, depth=altura_total, location=(0, 0, altura_total / 2))
poste = bpy.context.object

# Crear los escalones
for i in range(numero_escalones):
    bpy.ops.mesh.primitive_circle_add(radius=radio_externo, location=(0, 0, altura_total / 2 - (i + 1) * altura_escalon))
    escalon = bpy.context.object
    escalon.name = f"Escalon_{i}"
    bpy.ops.object.transform_apply(location=False, rotation=True, scale=False)
    bpy.ops.mesh.inset(thickness=radio_interno - radio_externo, inside=False)
    bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_shrink_fatten, value=-altura_escalon / 2)
    bpy.ops.object.transform_apply(location=False, rotation=True, scale=False)
