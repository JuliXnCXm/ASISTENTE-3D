 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Función para crear un peldaño de hormigón
def create_step(location):
    bpy.ops.mesh.primitive_cube_add(size=0.1, location=location)
    step = bpy.context.object
    step.scale.x = 2.5
    step.scale.y = 0.1
    step.scale.z = 0.1
    return step

# Función para crear la baranda metálica
def create_railing(location):
    bpy.ops.mesh.primitive_cube_add(size=0.05, location=location)
    railing = bpy.context.object
    railing.scale.x = 2.5
    railing.scale.y = 0.1
    railing.scale.z = 0.05
    return railing

# Crear la escalera
steps = []
for i in range(15):
    step_location = (i * 0.3, 0, -i * 0.3)
    steps.append(create_step(step_location))

# Crear la baranda en el descanso superior
railing_location = (7.5, 0, -4.5)
create_railing(railing_location)
```
