 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el sofá
bpy.ops.mesh.primitive_cube_add(size=2, location=(0, 0, 0))
sofa = bpy.context.object
sofa.name = "Sofa"

# Añadir modificador de subdivisión para suavizar el sofá
bpy.ops.object.modifier_add(type='SUBSURF')
subsurf = sofa.modifiers["Subdivision"]
subsurf.levels = 2
subsurf.render_levels = 2

# Crear la mesa de centro
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -1, 0))
table = bpy.context.object
table.name = "Table"

# Añadir modificador de subdivisión para suavizar la mesa
bpy.ops.object.modifier_add(type='SUBSURF')
subsurf = table.modifiers["Subdivision"]
subsurf.levels = 2
subsurf.render_levels = 2

# Cambiar el color del sofá a gris
sofa.data.materials.append(bpy.data.materials.new(name="GrayMaterial"))
sofa.data.materials[0].diffuse_color = (0.5, 0.5, 0.5)

# Cambiar el color de la mesa a madera
table.data.materials.append(bpy.data.materials.new(name="WoodMaterial"))
table.data.materials[0].diffuse_color = (0.8, 0.4, 0.2)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe
import os
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
