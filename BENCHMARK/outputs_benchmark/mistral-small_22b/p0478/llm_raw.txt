 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Parámetros del edificio
ancho = 20  # metros
altura = 5  # pisos
num_ventanas = 10  # número de ventanas por piso
espacio_ventana = ancho / num_ventanas  # espacio entre ventanas

# Crear el edificio
bpy.ops.mesh.primitive_cube_add(size=1, location=(ancho/2, 0, altura))
edificio = bpy.context.object

# Añadir modificador de subdivisión para suavizar el edificio
bpy.ops.object.modifier_add(type='SUBSURF')
subsurf = edificio.modifiers["Subsurf"]
subsurf.levels = 2
subsurf.render_levels = 2

# Crear ventanas
for piso in range(altura):
    for i in range(num_ventanas):
        bpy.ops.mesh.primitive_cube_add(size=0.1, location=(i * espacio_ventana + (espacio_ventana / 2), 0, piso))
        ventana = bpy.context.object
        ventana.scale.z = 4  # Altura de la ventana

# Guardar el archivo si BLEND_OUT está definido
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
