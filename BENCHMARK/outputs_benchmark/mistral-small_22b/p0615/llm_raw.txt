 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear suelo de porcelanato
bpy.ops.mesh.primitive_plane_add(size=8, location=(0, 0, 0))
floor = bpy.context.object
floor.name = "Floor"

# Crear material para el suelo
material = bpy.data.materials.new(name="Porcelanato")
material.diffuse_color = (1, 1, 1)
floor.data.materials.append(material)

# Zona de estar
sofa = bpy.ops.mesh.primitive_cube_add(size=2, location=(3, 0, 0))
shelf = bpy.ops.mesh.primitive_cube_add(size=1, location=(4, -1, 1))

# Zona de comedor
table = bpy.ops.mesh.primitive_cylinder_add(radius=0.5, depth=1.2, location=(3, -3, 0))
for i in range(4):
    chair = bpy.ops.mesh.primitive_cube_add(size=0.8, location=(i % 2 - 1 + 3, -int(i / 2) - 3, 0))

# Luz de área en el techo
bpy.ops.object.light_add(type='AREA', location=(4, 3, 3), size=6)
area_light = bpy.context.object
area_light.data.energy = 2000

# Guardar .blend si BLEND_OUT está definido
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
