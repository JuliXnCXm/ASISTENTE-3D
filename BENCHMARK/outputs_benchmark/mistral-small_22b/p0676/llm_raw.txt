 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crea el sofá
def create_sofa():
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    sofa = bpy.context.object
    sofa.name = "Sofa"

    # Añadir material gris
    mat = bpy.data.materials.new(name="GrayMaterial")
    mat.diffuse_color = (0.5, 0.5, 0.5)
    sofa.data.materials.append(mat)

    # Escalar el sofá para que sea más largo y bajo
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    sofa.scale = (1, 2, 0.5)

create_sofa()

# Crea la mesa de centro
def create_table():
    bpy.ops.mesh.primitive_cube_add(size=1, location=(1, 0, -0.7))
    table = bpy.context.object
    table.name = "Table"

    # Añadir material de madera de roble
    mat = bpy.data.materials.new(name="WoodMaterial")
    mat.diffuse_color = (0.8, 0.6, 0.4)
    table.data.materials.append(mat)

    # Escalar la mesa para que sea más larga y baja
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    table.scale = (1, 2, 0.5)

create_table()

# Guardar el archivo .blend si BLEND_OUT está definido
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
