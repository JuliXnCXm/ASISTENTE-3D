 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el sofá
def create_sofa():
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    sofa = bpy.context.object
    sofa.name = "Sofa"
    sofa.scale = (3, 1, 1)

    # Crear material de tela gris
    mat_sofa = bpy.data.materials.new(name="GrayCloth")
    mat_sofa.diffuse_color = (0.5, 0.5, 0.5)
    sofa.data.materials.append(mat_sofa)

# Crear la mesa de centro
def create_table():
    bpy.ops.mesh.primitive_cube_add(size=1, location=(2, 0, 0))
    table = bpy.context.object
    table.name = "Table"
    table.scale = (1, 3, 1)

    # Crear material de madera de nogal
    mat_table = bpy.data.materials.new(name="WalnutWood")
    mat_table.diffuse_color = (0.5, 0.25, 0.1)
    table.data.materials.append(mat_table)

# Crear el sofá y la mesa de centro
create_sofa()
create_table()

# Guardar el archivo .blend si existe BLEND_OUT
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
