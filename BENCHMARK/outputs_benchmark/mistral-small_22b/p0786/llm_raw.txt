 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el marco de la cama
frame = bpy.data.objects.new("Frame", None)
bpy.context.collection.objects.link(frame)

# Crear los postes del marco
post1 = bpy.data.objects.new("Post1", None)
post2 = bpy.data.objects.new("Post2", None)
post3 = bpy.data.objects.new("Post3", None)
post4 = bpy.data.objects.new("Post4", None)

frame.objects.link(post1)
frame.objects.link(post2)
frame.objects.link(post3)
frame.objects.link(post4)

# Posicionar los postes
post1.location = (0, 0.8, 0)
post2.location = (1.6, 0.8, 0)
post3.location = (0, 0.8, 2)
post4.location = (1.6, 0.8, 2)

# Crear el colchón
colchon = bpy.data.objects.new("Colchon", None)
bpy.context.collection.objects.link(colchon)

# Posicionar el colchón
colchon.location = (0.8, 1, 1)

# Asignar materiales
material_frame = bpy.data.materials.new(name="Wood")
material_frame.diffuse_color = (0.5, 0.35, 0.35, 1)

material_colchon = bpy.data.materials.new(name="Textile")
material_colchon.diffuse_color = (1, 1, 1, 1)

# Asignar materiales a los objetos
post1.data.materials.append(material_frame)
post2.data.materials.append(material_frame)
post3.data.materials.append(material_frame)
post4.data.materials.append(material_frame)
colchon.data.materials.append(material_colchon)

# Guardar el archivo si la variable de entorno BLEND_OUT está definida
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
