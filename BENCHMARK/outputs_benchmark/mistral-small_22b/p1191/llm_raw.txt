 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el marco de la cama
frame = bpy.data.objects.new("Frame", None)
bpy.context.collection.objects.link(frame)

# Crear los bordes del marco
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
cube = bpy.context.object
cube.scale.x = 2.0 / 2
cube.scale.y = 0.15
cube.scale.z = 1.6 / 2
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
frame.modifiers.new("Mirror", type='MIRROR')
frame.modifiers["Mirror"].mirror_plane = 'XY'
cube.select_set(True)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.extrude_region_move(TRANSFORM_OT_shrink_fatten={"value":0.1})
bpy.ops.object.mode_set(mode='OBJECT')
cube.select_set(False)
frame.modifiers["Mirror"].enabled = False

# Crear el colchón
colchon = bpy.data.objects.new("Colchon", None)
bpy.context.collection.objects.link(colchon)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
colchon = bpy.context.object
colchon.scale.x = 2.0
colchon.scale.y = 0.3
colchon.scale.z = 1.6
bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

# Asignar materiales
materials = bpy.data.materials
if not materials.get("Robe"):
    mat_roble = materials.new(name="Robe")
    mat_roble.diffuse_color = (0.5, 0.3, 0.1)
if not materials.get("Textil_Blanco"):
    mat_textil = materials.new(name="Textil_Blanco")
    mat_textil.diffuse_color = (1, 1, 1)
frame.data.materials.append(mat_roble)
colchon.data.materials.append(mat_textil)

# Guardar el archivo si BLEND_OUT está definido
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
