 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear terreno de césped
terreno = bpy.data.objects.new("Terreno", None)
bpy.context.collection.objects.link(terreno)
terreno.location = (0, 0, 0)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.primitive_plane_add(size=1)
bpy.context.object.scale = (25, 20, 1)
bpy.ops.object.shade_smooth()
bpy.ops.object.mode_set(mode='OBJECT')
terreno.select_set(True)
bpy.ops.object.material_slot_add()
material = bpy.data.materials.new(name="Césped")
material.diffuse_color = (0, 1, 0, 1)
terreno.data.materials[0] = material
bpy.ops.object.select_all(action='DESELECT')

# Crear casa de campo
casa = bpy.data.objects.new("Casa", None)
bpy.context.collection.objects.link(casa)
casa.location = (0, 0, 0)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.primitive_cube_add(size=1)
bpy.context.object.scale = (12, 9, 3)
bpy.ops.object.shade_smooth()
bpy.ops.object.mode_set(mode='OBJECT')
casa.select_set(True)
material = bpy.data.materials.new(name="Casa")
material.diffuse_color = (0.8, 0.7, 0.5, 1)
casa.data.materials[0] = material
bpy.ops.object.select_all(action='DESELECT')

# Crear tejado a dos aguas
tejado = bpy.data.objects.new("Tejado", None)
bpy.context.collection.objects.link(tejado)
tejado.location = (0, 0, 3)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.primitive_cone_add(vertices=4, radius1=6, depth=2, end_fill_type='NOTHING')
bpy.context.object.rotation_euler = (0, 0, 0.785)
bpy.ops.object.shade_smooth()
bpy.ops.object.mode_set(mode='OBJECT')
tejado.select_set(True)
material = bpy.data.materials.new(name="Tejado")
material.diffuse_color = (0.5, 0.3, 0.1, 1)
tejado.data.materials[0] = material
bpy.ops.object.select_all(action='DESELECT')

# Crear árbol en el jardín frontal
arbol = bpy.data.objects.new("Árbol", None)
bpy.context.collection.objects.link(arbol)
arbol.location = (10, -5, 0)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=0.5, depth=4, end_fill_type='NOTHING', location=(0, 0, 0))
bpy.context.object.rotation_euler = (0, 0, 1.57)
bpy.ops.object.shade_smooth()
bpy.ops.object.mode_set(mode='OBJECT')
arbol.select_set(True)
material = bpy.data.materials.new(name="Árbol")
material.diffuse_color = (0, 0.5, 0, 1)
arbol.data.materials[0] = material
bpy.ops.object.select_all(action='DESELECT')

# Guardar el archivo .blend si la variable de entorno BLEND_OUT está definida
import os
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
