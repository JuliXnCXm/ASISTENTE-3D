 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Función para crear una columna
def create_column(location):
    bpy.ops.mesh.primitive_cylinder_add(radius=0.2, depth=3.0, location=location)

# Función para crear una ventana
def create_window(location, size=(1.5, 0.8)):
    bpy.ops.mesh.primitive_plane_add(size_x=size[0], size_y=size[1], location=location)

# Crear el edificio de apartamentos
for i in range(4):
    for j in range(15):
        for k in range(12):
            # Crear columnas en las esquinas
            if (j == 0 or j == 14) and (k == 0 or k == 11):
                create_column((j * 3, i * 3 + 1.5, k * 3))

            # Crear ventanas en todas las fachadas
            if (j != 0 and j != 14) or (k != 0 and k != 11):
                create_window((j * 3, i * 3 + 2.5, k * 3), size=(1.5, 0.8))
```
