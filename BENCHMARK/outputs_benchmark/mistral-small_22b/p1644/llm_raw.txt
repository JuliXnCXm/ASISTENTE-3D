 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el edificio
def create_building():
    # Dimensiones del edificio
    width = 15
    depth = 20
    height = 5 * 4  # 4 metros por piso

    # Crear el suelo
    bpy.ops.mesh.primitive_cube_add(size=1, location=(width / 2, depth / 2, 0))
    floor = bpy.context.object
    floor.scale = (width, depth, 0.01)

    # Crear las paredes y el techo
    for i in range(5):
        bpy.ops.mesh.primitive_cube_add(size=1, location=(width / 2, depth / 2, height - i * 4))
        wall = bpy.context.object
        wall.scale = (width, 0.01, 4)
        if i == 0:
            wall.location.z += 0.5
        elif i == 4:
            wall.scale = (width, depth, 0.01)

    # Crear las columnas estructurales
    for x in range(0, width, 3):
        for z in range(0, depth, 3):
            bpy.ops.mesh.primitive_cylinder_add(radius=0.5, location=(x + 1.5, z + 1.5, height))
            column = bpy.context.object
            column.scale = (0.2, 0.2, height)

    # Crear la fachada de cristal
    for x in range(0, width, 3):
        for z in range(0, depth, 3):
            bpy.ops.mesh.primitive_plane_add(size=1, location=(x + 1.5, z + 1.5, height - 2))
            glass = bpy.context.object
            glass.scale = (3, 3, 0.01)

    # Crear la plaza de acceso
    bpy.ops.mesh.primitive_cube_add(size=1, location=(width / 2 + 5, depth / 2 - 5, 0))
    plaza = bpy.context.object
    plaza.scale = (10, 10, 0.01)

create_building()
```
