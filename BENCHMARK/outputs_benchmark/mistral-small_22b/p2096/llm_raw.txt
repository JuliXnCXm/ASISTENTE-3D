 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crea el dormitorio principal
bed = bpy.data.objects.new("Bed", None)
bed.location = (0, 0, 0)
bed.scale = (2, 1, 0.5)  # Ancho, largo, altura en metros
bpy.context.collection.objects.link(bed)

# Crea el cabecero tapizado
headboard = bpy.data.objects.new("Headboard", None)
headboard.location = (0, 1.5, 2)  # Posición relativa a la cama
headboard.scale = (2, 0.5, 0.5)  # Ancho, largo, altura en metros
bpy.context.collection.objects.link(headboard)

# Crea el muro bajo separador para el vestidor
divider_wall = bpy.data.objects.new("DividerWall", None)
divider_wall.location = (3, 0, 0)  # Posición en metros
divider_wall.scale = (1, 2, 0.5)  # Ancho, largo, altura en metros
bpy.context.collection.objects.link(divider_wall)

# Crea los armarios empotrados
wardrobe = bpy.data.objects.new("Wardrobe", None)
wardrobe.location = (4, 0, 0)  # Posición en metros
wardrobe.scale = (1, 2, 2)  # Ancho, largo, altura en metros
bpy.context.collection.objects.link(wardrobe)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe
import os
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
