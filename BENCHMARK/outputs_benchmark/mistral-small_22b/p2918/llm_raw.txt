 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crea el muro perimetral
bpy.ops.mesh.primitive_cube_add(size=0.2, location=(5, 0, 1.25))
muro = bpy.context.object

# Añade la apertura para el portón
bpy.ops.mesh.primitive_plane_add(size=3, location=(5, 0, 0))
porton = bpy.context.object

# Alinea el portón con el muro
porton.rotation_euler = (1.5708, 0, 0)

# Une los objetos en uno solo
bpy.ops.object.select_all(action='DESELECT')
muro.select_set(True)
porton.select_set(True)
bpy.context.view_layer.objects.active = muro
bpy.ops.object.join()

# Ajusta la ubicación del muro unido
muro.location = (5, 0, 1.25)

# Crea el muro opuesto
bpy.ops.mesh.primitive_cube_add(size=0.2, location=(5, 0, -1.25))
muro_opuesto = bpy.context.object

# Une los objetos en uno solo
bpy.ops.object.select_all(action='DESELECT')
muro.select_set(True)
muro_opuesto.select_set(True)
bpy.context.view_layer.objects.active = muro
bpy.ops.object.join()

# Ajusta la ubicación del muro unido
muro.location = (5, 0, 1.25)

# Guardar el archivo si BLEND_OUT está definido
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
