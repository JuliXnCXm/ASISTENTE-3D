 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear la base de la cubierta
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
base = bpy.context.object
base.scale = (8, 12, 0.25)

# Crear la cumbrera
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 3))
cumbrera = bpy.context.object
cumbrera.scale = (8, 12, 0.5)

# Crear las pendientes
bpy.ops.mesh.primitive_cone_add(vertices=4, radius1=6, depth=3, location=(0, 0, 3))
pendiente = bpy.context.object
pendiente.scale = (8, 12, 0.5)
pendiente.rotation_euler = (0, 0, 0.7854)  # 45 grados

# Crear la otra pendiente
bpy.ops.object.duplicate(linked=True)
otra_pendiente = bpy.context.object
otra_pendiente.location = (0, 0, -3)
otra_pendiente.rotation_euler = (0, 0, -0.7854)  # -45 grados

# Guardar el archivo si la variable de entorno BLEND_OUT existe

    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
