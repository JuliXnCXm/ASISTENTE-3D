 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear la base de la cubierta
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
base = bpy.context.object
base.scale = (8, 12, 0.25)

# Crear la cumbrera
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 3))
cumbrera = bpy.context.object
cumbrera.scale = (8, 12, 0.5)

# Crear las pendientes
bpy.ops.mesh.primitive_cone_add(vertices=4, radius1=6, depth=3, location=(0, 0, 3))
pendiente = bpy.context.object
pendiente.rotation_euler = (1.5708, 0, 0)
pendiente.scale = (1, 1, 0.25)

# Crear la otra pendiente
bpy.ops.mesh.primitive_cone_add(vertices=4, radius1=6, depth=3, location=(0, 0, 3))
pendiente2 = bpy.context.object
pendiente2.rotation_euler = (1.5708, 0, 0)
pendiente2.scale = (1, 1, 0.25)
pendiente2.location = (0, 6, 3)

# Unir todos los objetos en uno solo
bpy.ops.object.select_all(action='DESELECT')
base.select_set(True)
cumbrera.select_set(True)
pendiente.select_set(True)
pendiente2.select_set(True)
bpy.context.view_layer.objects.active = base
bpy.ops.object.join()

# Guardar el archivo si la variable de entorno BLEND_OUT existe
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
