 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Función para crear un cubo
def create_cuboid(name, size_x, size_y, size_z):
    bpy.ops.mesh.primitive_cube_add(size=1)
    obj = bpy.context.object
    obj.name = name
    obj.scale = (size_x, size_y, size_z)

# Función para crear un cono
def create_cone(name, height, radius):
    bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=radius, depth=height)
    obj = bpy.context.object
    obj.name = name

# Función para crear un cilindro
def create_cylinder(name, height, radius):
    bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=radius, depth=height)
    obj.name = name

# Crear el espacio principal de la oficina
create_cuboid("Floor", 8, 6, 0.1)
create_cuboid("Ceiling", 8, 6, 2.7)
create_cuboid("Wall_Left", 0.1, 6, 2.8)
create_cuboid("Wall_Right", 0.1, 6, 2.8)
create_cuboid("Wall_Back", 8, 0.1, 2.8)
create_cuboid("Wall_Front", 8, 0.1, 2.8)

# Crear la sala de reuniones
create_cuboid("Reunion_Floor", 3, 4, 0.1)
create_cuboid("Reunion_Ceiling", 3, 4, 2.7)
create_cuboid("Reunion_Wall_Left", 0.1, 4, 2.8)
create_cuboid("Reunion_Wall_Right", 0.1, 4, 2.8)
create_cuboid("Reunion_Wall_Back", 3, 0.1, 2.8)
create_cuboid("Reunion_Wall_Front", 3, 0.1, 2.8)
create_cuboid("Glass_Wall_Left", 0.1, 4, 2.8)
create_cuboid("Glass_Wall_Back", 3, 0.1, 2.8)

# Crear los puestos de trabajo
for i in range(1, 5):
    create_cuboid(f"Desk_{i}", 1.5, 0.6, 0.7)
    create_cylinder(f"Chair_{i}_Base", 0.3, 0.4)
    create_cone(f"Chair_{i}_Seat", 0.2, 0.4)
    create_cuboid(f"Drawer_{i}", 1.5, 0.6, 0.3)

# Amueblar la sala de reuniones
create_cuboid("Reunion_Table", 3, 2, 0.7)
for i in range(1, 7):
    create_cylinder(f"Reunion_Chair_{i}_Base", 0.3, 0.4)
    create_cone(f"Reunion_Chair_{i}_Seat", 0.2, 0.4)

# Guardar el archivo si la variable de entorno BLEND_OUT está definida
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
