 ```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Función para crear una pared
def create_wall(location, rotation=(0, 0, 0), size=(2, 2, 0.1)):
    bpy.ops.mesh.primitive_cube_add(location=location, rotation=rotation)
    obj = bpy.context.object
    obj.scale = (size[0] / 2, size[1] / 2, size[2])
    return obj

# Función para crear una ventana
def create_window(location, size=(1, 1.5, 0)):
    bpy.ops.mesh.primitive_cube_add(location=location)
    obj = bpy.context.object
    obj.scale = (size[0] / 2, size[1] / 2, size[2])
    return obj

# Crear paredes y ventanas del apartamento
wall_locations = [
    (0, 0, 0), (0, 5.5, 0), (5.5, 5.5, 0), (5.5, 11, 0), (11, 11, 0), (11, 0, 0),
    (7, 0, 0), (7, 3, 0), (9, 3, 0), (9, 0, 0)
]
window_locations = [
    (2.5, 4, 0.1), (8.5, 4, 0.1), (9.5, 4, 0.1)
]

for loc in wall_locations:
    create_wall(location=loc)

for loc in window_locations:
    create_window(location=loc)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe
if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
