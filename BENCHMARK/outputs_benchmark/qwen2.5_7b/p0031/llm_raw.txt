```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el piso rectangular
bpy.ops.mesh.primitive_plane_add(size=6, enter_editmode=False, align='WORLD', location=(0, 0, 0))
floor = bpy.context.object

# Escalar para obtener el espesor de 0.20 metros
floor.scale = (1, 1, 0.2)

# Mover el piso al suelo
floor.location = (0, 0, -0.1)
```
