```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones del peldaño
peldaño_ancho = 1.0
peldaño_huella = 0.3
peldaño_contrahuella = 0.18

# Definir la cantidad de peldaños
num_peldaños = 15

# Crear el primer peldaño
bpy.ops.mesh.primitive_cube_add(size=peldaño_ancho, location=(0, 0, 0))
peldaño_objeto = bpy.context.object
peldaño_objeto.name = "Peldaño_1"

# Rotar y mover el primer peldaño para que esté en la posición correcta
bpy.ops.transform.rotate(value=-1.5708, orient_axis='X')
bpy.ops.transform.translate(value=(0, -peldaño_huella / 2, -peldaño_contrahuella))

# Crear los demás peldaños y moverlos en el eje Y para formar la escalera
for i in range(1, num_peldaños):
    nuevo_peldaño = peldaño_objeto.copy()
    nuevo_peldaño.location.y -= peldaño_huella + 0.18
    bpy.context.collection.objects.link(nuevo_peldaño)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe
if "BLEND_OUT" in os.environ:
    blend_out_path = os.environ["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
