```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones del muro
length = 4.0
height = 2.5
thickness = 0.15

# Crear el muro de partición interior
bpy.ops.mesh.primitive_cube_add(size=thickness, location=(length/2, height/2, thickness/2))
wall = bpy.context.object

# Ajustar la escala del muro para que tenga las dimensiones correctas
wall.scale = (length, height, thickness)

# Mover el muro al centro de la escena
wall.location -= (length / 2, height / 2, 0)
```
