```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones en metros
length = 2.0
width = 1.5
height = 0.8

# Crear el marco de la cama (estructura de madera)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
cama = bpy.context.object
cama.scale = (length / 2, width / 2, height / 2)

# Crear los patas de la cama
for i in range(4):
    bpy.ops.mesh.primitive_cube_add(size=0.1, location=(0, 0, -height / 2 + 0.1))
    pata = bpy.context.object
    pata.scale = (0.3, 0.3, height * 0.8)
    if i % 2 == 0:
        pata.location[0] += length / 4
    else:
        pata.location[0] -= length / 4

# Crear el colchón
colchon_size = (length - 0.5, width - 0.3, 0.1)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, height / 2))
colchon = bpy.context.object
colchon.scale = colchon_size

# Asignar materiales (simplificado)
material_madera = bpy.data.materials.new(name="Madera")
material_colchon = bpy.data.materials.new(name="Colchón")

material_madera.diffuse_color = (0.5, 0.4, 0.2, 1)  # Color madera
material_colchon.diffuse_color = (1, 1, 1, 1)  # Blanco

cama.data.materials.append(material_madera)
colchon.data.materials.append(material_colchon)

# Guardar el archivo .blend si la variable BLEND_OUT está definida
if "BLEND_OUT" in bpy.context.scene:
    bpy.ops.wm.save_as_mainfile(filepath=bpy.context.scene[BLEND_OUT])
```
