```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el marco de la cama (estructura de madera)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
cama_marco = bpy.context.object

# Ajustar el tamaño del marco para simular una cama matrimonial
cama_marco.scale = (2.5, 1.8, 0.1)

# Asignar un material de madera roble al marco
material_roble = bpy.data.materials.new(name="MaderaRoble")
material_roble.diffuse_color = (0.639, 0.247, 0.078, 1)
cama_marco.data.materials.append(material_roble)

# Crear el colchón
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -0.9, -0.05))
colchon = bpy.context.object

# Ajustar el tamaño del colchón
colchon.scale = (2.4, 1.7, 0.1)

# Asignar un material blanco al colchón
material_blanco = bpy.data.materials.new(name="Blanco")
material_blanco.diffuse_color = (1, 1, 1, 1)
colchon.data.materials.append(material_blanco)

# Posicionar el colchón sobre el marco de la cama
bpy.context.view_layer.objects.active = cama_marco
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.select_all(action='DESELECT')
bpy.ops.mesh.select_face_by_solidify_offset(offset=0.1)
bpy.ops.object.mode_set(mode='OBJECT')

# Ajustar la posición del colchón
colchon.location = (0, -0.9 + 0.05, 0)

if 'BLEND_OUT' in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ['BLEND_OUT'])
```
