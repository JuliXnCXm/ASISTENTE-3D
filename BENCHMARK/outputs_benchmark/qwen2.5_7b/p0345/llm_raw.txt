```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las propiedades de los materiales
concrete_material = bpy.data.materials.new(name="Concrete")
metal_material = bpy.data.materials.new(name="Metal")

# Asignar los materiales a los objetos
concrete_material.use_nodes = True
metal_material.use_nodes = True

# Crear la escalera
peldaños = 15
ancho_peldaño = 0.3
largo_peldaño = 0.2
altura_peldaño = 0.15
espacio_entre_peldaños = 0.01

for i in range(peldaños):
    peldaño = bpy.ops.mesh.primitive_cube_add(
        location=(i * (ancho_peldaño + espacio_entre_peldaños), -largo_peldaño / 2, altura_peldaño * i))
    peldaño_obj = bpy.context.object
    peldaño_obj.scale = (ancho_peldaño, largo_peldaño, altura_peldaño)
    peldaño_obj.data.materials.append(concrete_material)

# Crear la baranda lineal metálica en el descanso superior
bpy.ops.mesh.primitive_cube_add(
    location=(peldaños * (ancho_peldaño + espacio_entre_peldaños), -largo_peldaño / 2, altura_peldaño * peldaños))
baranda_obj = bpy.context.object
baranda_obj.scale = (0.1, largo_peldaño, 0.05)
baranda_obj.data.materials.append(metal_material)

# Guardar el archivo .blend si la variable BLEND_OUT está definida
if 'BLEND_OUT' in os.environ:
    blend_out_path = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
