```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el sofá
sofa = bpy.data.objects.new(name="Sofa", object_data=None)
bpy.context.collection.objects.link(sofa)

sofa_mesh = bpy.data.meshes.new(name="SofaMesh")
sofa_obj = bpy.data.objects.new("SofaObj", sofa_mesh)

# Definir la geometría del sofá (3 plazas, 1.5m x 0.8m x 0.9m)
vertices_sofa = [
    (-0.75, -0.4, 0), (-0.75, 0.4, 0), (-0.75, 0.4, 2), (-0.75, -0.4, 2),
    (0.75, -0.4, 0), (0.75, 0.4, 0), (0.75, 0.4, 2), (0.75, -0.4, 2),
    (-1.5, -0.4, 0), (-1.5, 0.4, 0), (-1.5, 0.4, 2), (-1.5, -0.4, 2),
    (1.5, -0.4, 0), (1.5, 0.4, 0), (1.5, 0.4, 2), (1.5, -0.4, 2)
]

edges_sofa = []

faces_sofa = [
    [0, 1, 3, 2], [4, 5, 7, 6], [8, 9, 11, 10],
    [12, 13, 15, 14], [1, 5, 7, 3], [2, 6, 8, 4],
    [9, 13, 15, 11], [10, 14, 12, 8]
]

sofa_mesh.from_pydata(vertices_sofa, edges_sofa, faces_sofa)
sofa_mesh.update()

# Asignar material al sofá
material_sofa = bpy.data.materials.new(name="SofaMaterial")
material_sofa.diffuse_color = (0.537, 0.537, 0.537, 1)  # Gris
sofa_obj.data.materials.append(material_sofa)

# Crear la mesa de centro
table = bpy.data.objects.new(name="Table", object_data=None)
bpy.context.collection.objects.link(table)

table_mesh = bpy.data.meshes.new(name="TableMesh")
table_obj = bpy.data.objects.new("TableObj", table_mesh)

# Definir la geometría de la mesa (1m x 0.5m x 0.75m)
vertices_table = [
    (-0.25, -0.25, 0), (-0.25, 0.25, 0), (-0.25, 0.25, 0.75), (-0.25, -0.25, 0.75),
    (0.25, -0.25, 0), (0.25, 0.25, 0), (0.25, 0.25, 0.75), (0.25, -0.25, 0.75)
]

edges_table = []

faces_table = [
    [0, 1, 3, 2], [4, 5, 7, 6], [0, 1, 5, 4], [2, 3, 7, 6]
]

table_mesh.from_pydata(vertices_table, edges_table, faces_table)
table_mesh.update()

# Asignar material a la mesa
material_table = bpy.data.materials.new(name="TableMaterial")
material_table.diffuse_color = (0.843, 0.5, 0.298, 1)  # Madera
table_obj.data.materials.append(material_table)

# Posicionar el sofá y la mesa
sofa.location = (-2, 0, 0)
table.location = (2, 0, 0.375)

if "BLEND_OUT" in os.environ:
    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
