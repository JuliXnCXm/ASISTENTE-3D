bpy.ops.wm.read_factory_settings(use_empty=True)

concrete_floor.name = "ConcreteFloor"
concrete_floor.scale = (5, 4, 1)

brick_wall.name = "BrickWall"
brick_wall.scale = (5, 0.4, 1)
