import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el terreno
bpy.ops.mesh.primitive_plane_add(size=30, location=(0, 0, -1))
terreno = bpy.context.object
terreno.scale = (25, 30, 1)

# Crear la casa de dos pisos
bpy.ops.mesh.primitive_cube_add(size=4, location=(-4, -2, 2.5))
casa_basement = bpy.context.object
casa_basement.scale = (8, 10, 3)
bpy.ops.transform.translate(value=(0, 0, -3))

bpy.ops.mesh.primitive_cube_add(size=4, location=(-4, -2, 6.5))
casa_first_floor = bpy.context.object
casa_first_floor.scale = (8, 10, 3)

bpy.ops.mesh.primitive_cube_add(size=4, location=(-4, -2, 9.5))
casa_second_floor = bpy.context.object
casa_second_floor.scale = (8, 10, 3)

# Crear el tejado a dos aguas
tejado = bpy.ops.archipack屋顶(orientation='A', size=8, depth=6, height=2.5, location=(-4, -2, 9.75))
bpy.context.object.scale = (10, 3, 1)

# Crear el árbol
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.5, location=(10, -5, 3))
arbol = bpy.context.object

# Guardar la escena si se proporciona BLEND_OUT
