import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones del edificio
ancho = 15.0
altura_piso = 3.0
num_pisos = 3
espacio_ventana = 2.0

# Crear el suelo
bpy.ops.mesh.primitive_plane_add(size=ancho, location=(0, 0, -altura_piso * num_pisos / 2))
suelo = bpy.context.object
suelo.scale = (1, 1, altura_piso)

# Función para crear ventanas
def crear_ventana(x, y):
    ventana = bpy.ops.mesh.primitive_cube_add(size=1.0, location=(x, y, -altura_piso * num_pisos / 2 + 2))
    return ventana

# Crear las ventanas en una retícula regular
for i in range(int(ancho / espacio_ventana)):
    for j in range(num_pisos):
        crear_ventana(i * espacio_ventana - ancho / 2, j * altura_piso)

# Ajustar la escena para que se vea bien
bpy.context.scene.camera.location = (ancho + 5, 0, num_pisos * altura_piso + 5)
bpy.context.scene.camera.rotation_euler = (1.57, 0, 0)  # Mirar hacia arriba

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe

    blend_out_path = os.environ['BLEND_OUT']
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
