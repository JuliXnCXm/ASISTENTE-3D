```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el sofá de tres plazas de tela gris
sofa = bpy.data.objects.new("Sofa", None)
bpy.context.collection.objects.link(sofa)

# Definir la geometría del sofá (simplificado)
sofa_mesh = bpy.data.meshes.new(name="SofaMesh")
sofa_object = bpy.data.objects.new("SofaObject", sofa_mesh)

verts_sofa = [
    (-1, -0.5, 0), (-1, 0.5, 0), (1, 0.5, 0), (1, -0.5, 0),
    (-1, -0.75, 0.2), (-1, 0.75, 0.2), (1, 0.75, 0.2), (1, -0.75, 0.2)
]
edges_sofa = []
faces_sofa = [
    [0, 1, 3, 2], 
    [4, 5, 7, 6],
    [0, 1, 5, 4],
    [2, 3, 7, 6]
]

sofa_mesh.from_pydata(verts_sofa, edges_sofa, faces_sofa)
sofa_mesh.update()

# Colocar el sofá en la posición deseada
sofa.location = (0, -1.5, 0)

# Crear la mesa de centro de madera de roble
table = bpy.data.objects.new("Table", None)
bpy.context.collection.objects.link(table)

# Definir la geometría de la mesa (simplificada)
table_mesh = bpy.data.meshes.new(name="TableMesh")
table_object = bpy.data.objects.new("TableObject", table_mesh)

verts_table = [
    (-0.5, -0.25, 0), (-0.5, 0.25, 0), (0.5, 0.25, 0), (0.5, -0.25, 0),
    (-0.3, -0.1, 0.1), (-0.3, 0.1, 0.1), (0.3, 0.1, 0.1), (0.3, -0.1, 0.1)
]
edges_table = []
faces_table = [
    [0, 1, 3, 2], 
    [4, 5, 7, 6],
    [0, 1, 5, 4],
    [2, 3, 7, 6]
]

table_mesh.from_pydata(verts_table, edges_table, faces_table)
table_mesh.update()

# Colocar la mesa en la posición deseada
table.location = (0, 1.5, 0)

# Aplicar materiales
bpy.context.view_layer.objects.active = sofa_object
bpy.ops.object.shade_smooth()
sofa_material = bpy.data.materials.new(name="SofaMaterial")
sofa_material.diffuse_color = (0.6, 0.6, 0.6)  # Gris
sofa_object.data.materials.append(sofa_material)

bpy.context.view_layer.objects.active = table_object
table_material = bpy.data.materials.new(name="TableMaterial")
table_material.diffuse_color = (0.85, 0.42, 0.17)  # Madera de roble
table_object.data.materials.append(table_material)

# Guardar el archivo .blend si la variable BLEND_OUT está definida
if "BLEND_OUT" in os.environ:
    blend_out_path = os.environ["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
