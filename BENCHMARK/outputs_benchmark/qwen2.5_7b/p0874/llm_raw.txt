```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el sofá de tres plazas de tela gris
sofa = bpy.data.objects.new("Sofa", None)
bpy.context.collection.objects.link(sofa)

# Definir los vértices del sofá (en metros)
sofa_vertices = [
    (-1, -0.5, 0), (-1, 0.5, 0), (-0.7, 0.8, 0),
    (-0.7, 1.3, 0), (0.7, 1.3, 0), (0.7, 0.8, 0),
    (1, 0.5, 0), (1, -0.5, 0)
]

# Definir las caras del sofá
sofa_faces = [
    [0, 1, 2], [2, 3, 4], [4, 5, 6], [6, 7, 0]
]

# Crear la geometría del sofá
mesh_sofa = bpy.data.meshes.new(name="SofaMesh")
sofa_mesh_obj = bpy.data.objects.new("Sofa", mesh_sofa)
bpy.context.collection.objects.link(sofa_mesh_obj)

mesh_sofa.from_pydata(sofa_vertices, [], sofa_faces)
mesh_sofa.update()

# Crear la mesa de centro de madera de nogal
table = bpy.data.objects.new("Table", None)
bpy.context.collection.objects.link(table)

# Definir los vértices de la mesa (en metros)
table_vertices = [
    (-0.5, -0.2, 0), (-0.5, 0.2, 0), (0.5, 0.2, 0),
    (0.5, -0.2, 0), (-0.5, -0.2, 0)
]

# Definir las caras de la mesa
table_faces = [
    [0, 1, 2], [2, 3, 4], [4, 0, 1]
]

# Crear la geometría de la mesa
mesh_table = bpy.data.meshes.new(name="TableMesh")
table_mesh_obj = bpy.data.objects.new("Table", mesh_table)
bpy.context.collection.objects.link(table_mesh_obj)

mesh_table.from_pydata(table_vertices, [], table_faces)
mesh_table.update()

# Posicionar la mesa en frente del sofá (1.5 metros de distancia)

    bpy.ops.wm.save_as_mainfile(filepath=os.environ["BLEND_OUT"])
```
