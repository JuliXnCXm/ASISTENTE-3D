import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el piso
bpy.ops.mesh.primitive_plane_add(size=13, location=(0, 0, -0.1))
bpy.context.object.scale = (4, 3.5, 1)
bpy.ops.object.shade_smooth()

# Pintar las paredes de gris
for i in range(2):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(i*4-2, -1.75, 0))
    bpy.context.object.scale = (4, 3.5, 2)
    bpy.ops.object.shade_smooth()
    bpy.ops.paint.weight_paint_toggle()

# Crear la cama doble
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -1.75, -0.05))
bpy.context.object.scale = (1.8, 0.9, 0.2)
bpy.ops.object.shade_smooth()

# Crear la estructura de madera para la cama
bpy.ops.mesh.primitive_cube_add(size=0.1, location=(0, -1.75, -0.1))
bpy.context.object.scale = (0.1, 0.9, 0.2)
bpy.ops.object.shade_smooth()

# Crear el armario de dos puertas
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 1.75, -0.05))
bpy.context.object.scale = (1.8, 0.9, 2)
bpy.ops.object.shade_smooth()

# Guardar el archivo .blend si la variable BLEND_OUT está definida

    bpy.ops.wm.save_as_mainfile(filepath=bpy.path.abspath("//" + BLEND_OUT))
