```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define el tamaño del terreno
terreno_size = 5

# Crea el terreno como un plano
bpy.ops.mesh.primitive_plane_add(size=terreno_size, location=(0, 0, -1))
plane = bpy.context.object
plane.name = "Terreno"

# Define la geometría del árbol (copa esférica)
def crear_arbol_copa_esferica():
    # Crea un esfera para representar la copa del árbol
    bpy.ops.mesh.primitive_uv_sphere_add(radius=1, location=(0, 0, 2))
    arbol = bpy.context.object
    arbol.name = "Arbol"
    
    # Ajusta la escala para que parezca un árbol
    arbol.scale = (0.5, 0.5, 3)

crear_arbol_copa_esferica()

# Verifica si existe la variable de entorno BLEND_OUT y guarda el archivo si es así
if "BLEND_OUT" in locals() or "BLEND_OUT" in globals():
    bpy.ops.wm.save_as_mainfile(filepath=BLEND_OUT)
```
