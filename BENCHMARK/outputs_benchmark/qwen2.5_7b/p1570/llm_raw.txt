```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el suelo
bpy.ops.mesh.primitive_plane_add(size=8, enter_editmode=False, align='WORLD', location=(0, 0, 0))
suelo = bpy.context.object
suelo.scale = (4, 3, 1)
suelo.name = "Suelo"

# Crear el tejado a dos aguas
tejado = bpy.ops.mesh.primitive_cylinder_add(radius=2, depth=1.5, enter_editmode=False, align='WORLD', location=(0, 0, 1.5))
bpy.context.object.name = "Tejado"

# Rotar el techo para formar una estructura a dos aguas
tejado_obj = bpy.context.object
tejado_obj.rotation_euler = (1.5708, 0, 1.5708)

# Ajustar la posición del techo
tejado_obj.location = (0, -0.25, 1.5)

# Crear los dos lados del tejado
bpy.ops.mesh.primitive_cylinder_add(radius=2, depth=1.5, enter_editmode=False, align='WORLD', location=(0, 3.25, 1.5))
lado_derecho = bpy.context.object
lado_derecho.rotation_euler = (1.5708, 0, -1.5708)
lado_derecho.location = (4, 3.25, 1.5)

bpy.ops.mesh.primitive_cylinder_add(radius=2, depth=1.5, enter_editmode=False, align='WORLD', location=(0, -3.25, 1.5))
lado_izquierdo = bpy.context.object
lado_izquierdo.rotation_euler = (1.5708, 0, 1.5708)
lado_izquierdo.location = (-4, -3.25, 1.5)

# Ajustar la escena para que el techo quede correctamente
bpy.ops.object.select_all(action='DESELECT')
tejado_obj.select_set(True)
lado_derecho.select_set(True)
lado_izquierdo.select_set(True)
suelo.select_set(True)
bpy.context.view_layer.objects.active = suelo

# Ajustar la escena para que el techo quede correctamente
bpy.ops.object.join()

# Guardar el archivo .blend si existe BLEND_OUT
if "BLEND_OUT" in bpy.context.scene:
    blend_out_path = bpy.context.scene["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
