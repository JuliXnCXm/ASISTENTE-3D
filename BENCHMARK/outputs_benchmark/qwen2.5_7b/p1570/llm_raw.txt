```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones en metros
ancho = 4.0
fondo = 3.0
altura_tejado = 2.5

# Crea los vértices del tejado a dos aguas
vertices = [
    (-ancho/2, -fondo/2, 0),
    (ancho/2, -fondo/2, 0),
    (ancho/2, fondo/2, altura_tejado),
    (-ancho/2, fondo/2, altura_tejado)
]

# Crea los caras del tejado
caras = [
    (0, 1, 3), 
    (1, 2, 3)
]

# Agrega la geometría al contexto de la escena
mesh = bpy.data.meshes.new(name="Tejado")
obj = bpy.data.objects.new("Tejado", mesh)
bpy.context.collection.objects.link(obj)

# Asigna los vértices y caras a la malla
mesh.from_pydata(vertices, [], caras)
mesh.update()

# Ajusta el origen del objeto al centro de masa
obj.location = (0, 0, altura_tejado / 2)

# Guarda el archivo .blend si existe BLEND_OUT en las variables de entorno
if "BLEND_OUT" in bpy.context.scene:
    bpy.ops.wm.save_as_mainfile(filepath=bpy.context.scene["BLEND_OUT"])
```
