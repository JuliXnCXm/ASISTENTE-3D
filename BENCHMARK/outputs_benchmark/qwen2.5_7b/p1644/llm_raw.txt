```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones del edificio
pisos = 5
ancho = 15
largo = 20

# Crea el suelo
bpy.ops.mesh.primitive_plane_add(size=30, location=(0, 0, -1))
suelo = bpy.context.object
suelo.scale = (ancho, largo, 0.1)

# Crea las columnas estructurales
for i in range(6):
    for j in range(5):
        columna_x = -largo / 2 + j * (largo / 5)
        columna_z = -ancho / 2 + i * (ancho / 6)
        bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=15, location=(columna_x, 0, columna_z))
        
# Crea la fachada de cristal
fachada = bpy.ops.mesh.primitive_cube_add(size=1, location=(-largo / 2 + largo / 4, 0, -ancho / 2 + ancho / 6))
bpy.context.object.scale = (largo / 2 - largo / 4, ancho / 2 - ancho / 6, pisos)
fachada_obj = bpy.context.object
fachada_obj.name = "FachadaCristal"
fachada_obj.data.materials.append(bpy.data.materials.new(name="Cristal"))

# Crea la plaza de acceso
bpy.ops.mesh.primitive_plane_add(size=30, location=(0, 0, -pisos - 1))
plaza = bpy.context.object
plaza.scale = (ancho + 5, largo + 5, 0.1)

    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
