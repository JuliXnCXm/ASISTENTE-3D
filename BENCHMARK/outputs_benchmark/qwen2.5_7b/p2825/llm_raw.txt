```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear un bolardo cilíndrico
radius = 0.1  # Radio del bolardo (mitad del diámetro)
height = 0.9  # Altura del bolardo

# Crear el objeto cilindro
bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=(0, 0, 0))

# Renombrar el objeto para claridad
cylinder_obj = bpy.context.object
cylinder_obj.name = "Bolardo_Hormigon"

# Ajustar la escala si es necesario (no en este caso, pero podría ser útil para otros objetos)
# cylinder_obj.scale = (1, 1, 1)

# Guardar el archivo .blend si la variable de entorno BLEND_OUT existe
import os
if "BLEND_OUT" in os.environ:
    blend_out_path = os.getenv("BLEND_OUT")
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
