```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear un bolardo cilíndrico
bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=0.8, location=(0, 0, 0))

# Renombrar el objeto al bolardo
cylinder = bpy.context.object
cylinder.name = "Bolardo"

# Aplicar la escala para que coincida con los metros
bpy.ops.object.select_all(action='DESELECT')
cylinder.select_set(True)
bpy.context.view_layer.objects.active = cylinder
bpy.ops.object.scale_apply()

# Verificar si existe el entorno BLEND_OUT y guardar el archivo si es así
if "BLEND_OUT" in bpy.context.scene:
    blend_out_path = bpy.context.scene["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path, check_existing=False)
```
