```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir los materiales
wood_material = bpy.data.materials.new(name="Wood")
concrete_material = bpy.data.materials.new(name="Concrete")

# Asignar propiedades de material
wood_material.diffuse_color = (0.5, 0.23, 0.14, 1)
concrete_material.diffuse_color = (0.2, 0.2, 0.2, 1)

# Crear losa de hormigón
bpy.ops.mesh.primitive_cube_add(size=0.3, location=(0, 0, -0.15))
losa = bpy.context.object
losa.scale = (4.0, 3.0, 0.15)
losa.data.materials.append(concrete_material)

# Crear postes verticales
for i in range(4):
    x, y = (-2.0 + i % 2 * 4.0, -1.5 + i // 2 * 3.0)
    bpy.ops.mesh.primitive_cube_add(size=0.15, location=(x, y, 2.35))
    poste = bpy.context.object
    poste.scale = (0.15, 0.15, 2.5)
    poste.data.materials.append(wood_material)

# Crear vigas horizontales
viga1 = bpy.ops.mesh.primitive_cube_add(size=0.3, location=(-2.0, -1.5, 2.4))
bpy.context.object.scale = (4.0, 0.3, 0.1)
viga1 = bpy.context.object
viga1.data.materials.append(wood_material)

viga2 = bpy.ops.mesh.primitive_cube_add(size=0.3, location=(2.0, -1.5, 2.4))
bpy.context.object.scale = (4.0, 0.3, 0.1)
viga2 = bpy.context.object
viga2.data.materials.append(wood_material)

viga3 = bpy.ops.mesh.primitive_cube_add(size=0.3, location=(-2.0, 1.5, 2.4))
bpy.context.object.scale = (4.0, 0.3, 0.1)
viga3 = bpy.context.object
viga3.data.materials.append(wood_material)

viga4 = bpy.ops.mesh.primitive_cube_add(size=0.3, location=(2.0, 1.5, 2.4))
bpy.context.object.scale = (4.0, 0.3, 0.1)
viga4 = bpy.context.object
viga4.data.materials.append(wood_material)

# Guardar el archivo .blend si existe la variable de entorno BLEND_OUT
if "BLEND_OUT" in os.environ:
    blend_out_path = os.environ["BLEND_OUT"]
    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
