```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir las dimensiones en metros
width = 5.0
depth = 10.0
height = 3.0

# Crear el suelo
floor = bpy.ops.mesh.primitive_plane_add(size=width * depth, location=(0, 0, 0))
bpy.context.object.scale = (width, depth, height)

# Crear la pared frontal y trasera
front_wall = bpy.ops.mesh.primitive_cube_add(location=(-width / 2 + 1.5, -depth / 2 + 1.5, height / 2))
back_wall = bpy.ops.mesh.primitive_cube_add(location=(width / 2 - 1.5, -depth / 2 + 1.5, height / 2))

# Crear la pared lateral izquierda
left_wall = bpy.ops.mesh.primitive_cube_add(location=(-width / 2 + 1.5, depth / 2 - 1.5, height / 2))
left_wall[0].scale = (3.0, depth, height)

# Crear la pared lateral derecha
right_wall = bpy.ops.mesh.primitive_cube_add(location=(width / 2 - 1.5, depth / 2 - 1.5, height / 2))
right_wall[0].scale = (3.0, depth, height)

# Crear el techo
ceiling = bpy.ops.mesh.primitive_plane_add(size=width * depth, location=(0, 0, height))
bpy.context.object.scale = (width, depth, -height / 2)

# Crear la puerta principal
door_location = (-width / 2 + 1.5, -depth / 4 + 1.5, height / 2)
door_size = (3.0, 2.0, height)
bpy.ops.mesh.primitive_cube_add(location=door_location, scale=door_size)

# Crear la ventana
window_location = (-width / 2 + 1.5, depth / 4 - 1.5, height / 2)
window_size = (2.0, 3.0, height)
bpy.ops.mesh.primitive_cube_add(location=window_location, scale=window_size)

# Crear la sala-comedor y cocina
living_room_cooking_area = bpy.ops.mesh.primitive_cube_add(location=(-width / 4 + 1.5, -depth / 2 + 3.0, height / 2))
living_room_cooking_area[0].scale = (3.0, 6.0, height)

# Crear el dormitorio principal
bedroom_location = (width / 4 - 1.5, -depth / 2 + 3.0, height / 2)
bedroom_size = (3.0, 4.0, height)
bpy.ops.mesh.primitive_cube_add(location=bedroom_location, scale=bedroom_size)

# Crear el baño
bathroom_location = (-width / 4 + 1.5, -depth / 2 + 6.0, height / 2)
bathroom_size = (3.0, 2.0, height)
bpy.ops.mesh.primitive_cube_add(location=bathroom_location, scale=bathroom_size)

    bpy.ops.wm.save_as_mainfile(filepath=blend_out_path)
```
