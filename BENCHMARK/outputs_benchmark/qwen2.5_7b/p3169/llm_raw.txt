# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear el piso

floor = bpy.context.object

floor.name = "Floor"

bpy.ops.mesh.primitive_cube_add(size=2.8, location=(-4, -3, 2.8))
wall1 = bpy.context.object

    chair_obj = bpy.context.object
    chair_obj.scale = (0.5, 0.5, 0.5)
    chair_obj.name = f"Chair{i}"

for i in range(6):

    chair_obj = bpy.context.object
    chair_obj.scale = (0.5, 0.5, 0.5)
    chair_obj.name = f"MeetingChair{i}"

# Crear el tabique de vidrio para la sala de reuniones
