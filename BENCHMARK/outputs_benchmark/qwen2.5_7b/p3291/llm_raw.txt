```python
import bpy

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Define las dimensiones en metros
edificio_altura = 10
edificio_ancho = 20
edificio_largo = 30
suelo_deck_madera_tam = 5
jardinera_hormigon_tam = 1.5
pergola_acero_tam = 4

# Crea el edificio
bpy.ops.mesh.primitive_cube_add(location=(0, 0, -edificio_altura / 2))
edificio = bpy.context.object
edificio.scale = (edificio_largo, edificio_ancho, edificio_altura)

# Crea el suelo de deck de madera
suelo_deck_madera = bpy.ops.mesh.primitive_plane_add(location=(0, 0, -edificio_altura / 2 - 1))
bpy.context.object.scale = (suelo_deck_madera_tam * edificio_largo, suelo_deck_madera_tam * edificio_ancho, 0.1)

# Crea las jardineras perimetrales de hormigón
for x in [-edificio_ancho / 2 + suelo_deck_madera_tam / 2, edificio_ancho / 2 - suelo_deck_madera_tam / 2]:
    for z in [-edificio_largo / 2 + suelo_deck_madera_tam / 2, edificio_largo / 2 - suelo_deck_madera_tam / 2]:
        jardinera = bpy.ops.mesh.primitive_cube_add(location=(x, 0, -edificio_altura / 2 - 1))
        jardinera_obj = bpy.context.object
        jardinera_obj.scale = (jardinera_hormigon_tam, suelo_deck_madera_tam * 3, jardinera_hormigon_tam)
        jardinera_obj.rotation_euler[0] = 1.57

# Crea la pérgola de acero
pergola_acero = bpy.ops.mesh.primitive_cube_add(location=(0, 0, -edificio_altura / 2 + 3))
bpy.context.object.scale = (pergola_acero_tam * edificio_largo, pergola_acero_tam * suelo_deck_madera_tam, pergola_acero_tam)
pergola_acero_obj = bpy.context.object
pergola_acero_obj.rotation_euler[0] = 1.57

# Ajusta la escena para que se vea bien en el render
bpy.context.scene.camera.location = (edificio_largo / 2 + 5, -edificio_ancho / 2 - 5, edificio_altura / 2 + 3)
bpy.context.scene.camera.rotation_euler = (1.0472, 0, 0)
```
