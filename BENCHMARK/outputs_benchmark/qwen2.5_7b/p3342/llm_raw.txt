```python
import bpy

# Limpiar la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Definir los muros y lasos en metros
wall_thickness = 0.25
floor_thickness = 0.1

# Zona de día (sala, comedor, cocina)
day_area_length = 10
day_area_width = 8
day_area_height = wall_thickness

# Zona de noche (dormitorios y baño)
night_area_length = 6
night_area_width = 5
night_area_height = wall_thickness

# Patio central
patio_length = 4
patio_width = 3
patio_height = floor_thickness

# Crear los muros y lasos para la zona de día
bpy.ops.mesh.primitive_cube_add(size=day_area_length, location=(0, -1.5 * day_area_width / 2, 0))
bpy.ops.transform.resize(value=(day_area_length, day_area_width, day_area_height))

bpy.ops.mesh.primitive_cube_add(size=day_area_length, location=(-1.5 * day_area_length / 2, -1.5 * day_area_width / 2, 0))
bpy.ops.transform.resize(value=(day_area_length, day_area_width, day_area_height))

bpy.ops.mesh.primitive_cube_add(size=day_area_width, location=(-1.5 * day_area_length / 2, -1.5 * day_area_width / 2 + day_area_width / 2, 0))
bpy.ops.transform.resize(value=(day_area_width, day_area_length, day_area_height))

bpy.ops.mesh.primitive_cube_add(size=day_area_width, location=(-1.5 * day_area_length / 2 - day_area_width / 2, -1.5 * day_area_width / 2 + day_area_width / 2, 0))
bpy.ops.transform.resize(value=(day_area_width, day_area_length, day_area_height))

# Crear los muros y lasos para la zona de noche
bpy.ops.mesh.primitive_cube_add(size=night_area_length, location=(-1.5 * (day_area_length + night_area_length) / 2, -1.5 * night_area_width / 2, 0))
bpy.ops.transform.resize(value=(night_area_length, night_area_width, night_area_height))

bpy.ops.mesh.primitive_cube_add(size=night_area_length, location=(-1.5 * (day_area_length + night_area_length) / 2 - night_area_length / 2, -1.5 * night_area_width / 2, 0))
bpy.ops.transform.resize(value=(night_area_length, night_area_width, night_area_height))

bpy.ops.mesh.primitive_cube_add(size=night_area_width, location=(-1.5 * (day_area_length + night_area_length) / 2 - night_area_length / 2, -1.5 * night_area_width / 2 + night_area_width / 2, 0))
bpy.ops.transform.resize(value=(night_area_width, night_area_length, night_area_height))

# Crear el patio central
bpy.ops.mesh.primitive_cube_add(size=patio_length, location=(-1.5 * (day_area_length + night_area_length) / 2 - night_area_length / 4, -1.5 * day_area_width / 2 + day_area_width / 2 - night_area_width / 4, floor_thickness / 2))
bpy.ops.transform.resize(value=(patio_length, patio_width, patio_height))

# Crear el mobiliario
# Sofá en la sala de estar
sofa = bpy.ops.mesh.primitive_cube_add(size=1.5, location=(-3, -6, 0.5))
bpy.context.object.name = "Sofa"

# Isla de cocina
island = bpy.ops.mesh.primitive_cube_add(size=2, location=(4, -8, 0.5))
bpy.context.object.name = "Island"

# Mesa de comedor
table = bpy.ops.mesh.primitive_cube_add(size=1.5, location=(-3, -9, 0.5))
bpy.context.object.name = "Table"

# Camas en los dormitorios
bed1 = bpy.ops.mesh.primitive_cube_add(size=2, location=(4, -6, 0.5))
bpy.context.object.name = "Bed1"
bed2 = bpy.ops.mesh.primitive_cube_add(size=2, location=(4, -7, 0.5))
bpy.context.object.name = "Bed2"
bed3 = bpy.ops.mesh.primitive_cube_add(size=2, location=(4, -8, 0.5))
bpy.context.object.name = "Bed3"

# Baño
bathroom = bpy.ops.mesh.primitive_cube_add(size=2, location=(-1, -6, 0.5))
bpy.context.object.name = "Bathroom"
```
