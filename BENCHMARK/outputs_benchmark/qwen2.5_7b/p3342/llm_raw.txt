import math

# Limpia la escena
bpy.ops.wm.read_factory_settings(use_empty=True)

# Crear las losas
bpy.ops.mesh.primitive_cube_add(size=1, location=(largo_zona_dia / 2 - ancho_zona_dia / 2 + muro_ancho / 2, largo_patio / 2 - ancho_patio / 2 + muro_ancho / 2, 0))

bpy.ops.mesh.primitive_cube_add(size=1, location=(largo_zona_noche / 2 - ancho_zona_noche / 2 + muro_ancho * 2 + 5.5, largo_patio / 2 - ancho_patio / 2 + muro_ancho * 2 + 3.5, 0.3))
